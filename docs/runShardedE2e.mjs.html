function pickPosInt(v, def, key, probs) {
    if (v === undefined || v === null) {
        return def
    }
    let n = typeof v === 'string' && v.trim() !== '' ? Number(v) : v

//讀設定檔(可不存在); 未知之鍵即拋錯: 拼錯 officialOnly 等於靜默丟掉限制
function readShardConfig(fp) {
    if (!fs.existsSync(fp)) {
        return { exists: false, cfg: {} }
    }
    let cfg
    try {
        cfg = JSON.parse(fs.readFileSync(fp, 'utf8'))
    }
    catch (err) {
        throw new Error(`runShardedE2e: 設定檔 ${fp} 無法解析(${err.message})`)
    }
    if (!isObj(cfg)) {
        throw new Error(`runShardedE2e: 設定檔 ${fp} 須為物件`)
    }
    let unknown = Object.keys(cfg).filter((k) => !CONFIG_KEYS.includes(k))
    if (unknown.length > 0) {
        throw new Error(`runShardedE2e: 設定檔 ${fp} 含未知之鍵: ${unknown.join(', ')}(可用: ${CONFIG_KEYS.join(', ')})`)
    }
    return { exists: true, cfg }
}


//檢查基準埠陣列
function checkPorts(ports) {
    if (!Array.isArray(ports)) {
        throw new Error('runShardedE2e: ports 須為基準埠陣列')
    }
    let list = ports.map(Number)
    let bad = list.filter((p) => !Number.isInteger(p) || p < 1 || p > 65535)
    if (bad.length > 0) {
        throw new Error(`runShardedE2e: ports 含不合法之埠: ${bad.join(', ')}`)
    }
    return [...new Set(list)]
}


//算各組之實際埠, 並擋掉跨組重疊(例: ports [8080,8180] 且 portStep 100)與溢位
function buildPortTable(groups, ports, portStep) {
    let table = []
    let seen = new Map()
    for (let g of groups) {
        let offset = g.index * portStep
        for (let base of ports) {
            let port = base + offset
            if (port > 65535) {
                throw new Error(`runShardedE2e: 第 ${g.index} 組之埠 ${base}+${offset}=${port} 超過 65535`)
            }
            if (seen.has(port)) {
                throw new Error(`runShardedE2e: 埠 ${port} 同時為第 ${seen.get(port)} 組與第 ${g.index} 組所用, 請調整 ports 或 portStep`)
            }
            seen.set(port, g.index)
            table.push({ group: g.index, base, offset, port })
        }
    }
    return table
}


//試綁定一個埠一次: 回 null 為成功, 否則回錯誤碼
function tryBind(port, host) {
    return new Promise((resolve) => {
        let s = net.createServer()
        s.once('error', (err) => resolve(err.code || 'ERROR'))
        s.listen({ port, host, exclusive: true }, () => {
            s.close(() => resolve(null))
        })
    })
}


//逐一位址試綁定: Windows 下特定位址與萬用位址可並存, 只試一種會漏掉占用者; 本機無 IPv6 者不算占用
async function checkPortFree(port) {
    for (let host of BIND_HOSTS) {
        let code = await tryBind(port, host)
        if (code === 'EADDRNOTAVAIL' || code === 'EAFNOSUPPORT') {
            if (host.includes(':')) {
                continue
            }
        }
        if (code !== null) {
            return { port, free: false, host, code }
        }
    }
    return { port, free: true, host: '', code: '' }
}


//等多個埠皆可綁定, 回傳仍不可綁定者
async function waitPortsFree(ports, timeoutMs) {
    let remain = ports
    let start = Date.now()
    while (true) {
        let rs = await Promise.all(remain.map((p) => checkPortFree(p)))
        let busy = rs.filter((r) => !r.free)
        if (busy.length === 0 || Date.now() - start >= timeoutMs) {
            return busy
        }
        remain = busy.map((r) => r.port)
        await sleep(300)
    }
}


//PID 之映像名稱(只取名稱, 不取命令列: 命令列可能含憑證)
function procName(pid, ctx = {}) {
    let run = ctx.exec || execDef
    try {
        if (process.platform === 'win32') {
            let r = run('tasklist', ['/FI', `PID eq ${pid}`, '/FO', 'CSV', '/NH'], { src: 'tasklist', timeout: 10000, maxBuffer: 1024 * 1024 })
            let m = /^"([^"]+)"/.exec(String(r.stdout || '').trim())
            return m ? m[1] : ''
        }
        let r = run('ps', ['-o', 'comm=', '-p', String(pid)], { src: 'tasklist', timeout: 10000, maxBuffer: 1024 * 1024 })
        return String(r.stdout || '').trim()
    }
    catch (err) {
        return ''
    }
}

/**
 * 取行程表：PID、父PID、起始時間與映像名，不取命令列(命令列可能含憑證)
 *
 * Windows之CreationDate同一行程每次相同(2026-11-04實測全部行程5次查詢1256筆比對0差異)，故可作身分比對；其他平台之lstart為秒級
 *
 * @param {Object} [opt={}] 輸入設定物件，預設{}
 * @param {Function} [opt.exec] 輸入同步執行子指令函數，格式為(prog, args, { src, timeout, maxBuffer })=>{ status, stdout, error }(與child_process.spawnSync同形)，src為'table'，預設為不經shell、隱藏視窗之spawnSync
 * @param {String} [opt.platform=process.platform] 輸入作業系統字串，預設process.platform
 * @returns {Array|null} 回傳行程陣列，元素為{pid,ppid,startMs,image}；查詢失敗或查無任何行程回傳null(呼叫端不得把null當成無行程)
 * @example
 *
 * import { listProcessTable } from 'w-package-tools-e2e/src/runShardedE2e.mjs'
 *
 * let table = listProcessTable()
 * console.log(table.find((p) => p.pid === process.pid).image)
 * // => 'node.exe'
 *
 */
export function listProcessTable(opt = {}) {
    let { exec, platform = process.platform } = opt
    let run = exec || execDef
    let o = { src: 'table', timeout: 30000, maxBuffer: 16 * 1024 * 1024 }
    let win = platform === 'win32'
    let r
    try {
        r = win
            ? run('powershell', ['-NoProfile', '-NonInteractive', '-Command', PS_TABLE], o)
            : run('ps', ['-A', '-o', 'pid=,ppid=,lstart=,comm='], o)
    }
    catch (err) {
        return null
    }
    if (!r || r.error || r.status !== 0) {
        return null
    }
    let re = win ? /^(\d+) (\d+) (\S+) ?(.*)$/ : /^\s*(\d+)\s+(\d+)\s+(\S+\s+\S+\s+\d+\s+[\d:]+\s+\d+)\s*(.*)$/
    let rows = String(r.stdout || '').split(/\r?\n/).map((l) => {
        let m = re.exec(win ? l.trim() : l)
        return m ? { pid: Number(m[1]), ppid: Number(m[2]), startMs: Date.parse(m[3]), image: m[4].trim() } : null
    }).filter((v) => v && Number.isFinite(v.startMs))
    return rows.length > 0 ? rows : null
}


//身分鍵: PID 與起始時間皆須完全相同
let keyOf = (p) => `${Number(p.pid)}@${p.startMs}`


//由種子沿存活之父子鏈走訪: 父自起始至今持續持有該 PID, 此期間建立且 ppid 為該 PID 者必為其子; 起始較早者為 PID 回收造成之假關係。回傳由深到淺
function walkLiveTree(table, seeds, protect = new Set()) {
    let found = new Map()
    let queue = []
    for (let { row, basis } of seeds) {
        if (!found.has(row.pid) && !protect.has(row.pid)) {
            let v = { row, basis, depth: 0 }
            found.set(row.pid, v)
            queue.push(v)
        }
    }
    while (queue.length > 0) {
        let p = queue.shift()
        for (let c of table) {
            if (c.ppid === p.row.pid && c.pid !== p.row.pid && !found.has(c.pid) && !protect.has(c.pid) && c.startMs >= p.row.startMs) {
                let v = { row: c, basis: `chain:${p.row.pid}`, depth: p.depth + 1 }
                found.set(c.pid, v)
                queue.push(v)
            }
        }
    }
    return [...found.values()].sort((a, b) => b.depth - a.depth)
}


/**
 * 由行程表找出身分已確認且仍存活之根行程，及沿存活父子鏈走到之全部子孫
 *
 * 根須PID相同且起始時間與紀錄完全相同；已結束之根不走其下：Windows之ParentProcessId於父結束後仍留在子行程上，且PID會回收，
 * 「父已結束＋ppid相同＋起始較晚」不足以判定為子孫(2026-11-04疑似誤殺他套件派工之成因)。子孫須起始不早於其存活之父
 *
 * @param {Array} table 輸入行程表陣列，元素為{pid,ppid,startMs}
 * @param {Array} roots 輸入根行程陣列，元素為{pid,startMs}
 * @param {Object} [opt={}] 輸入設定物件，預設{}
 * @param {Array} [opt.protect=[]] 輸入不納入且不往下走之PID陣列，預設[]
 * @returns {Array} 回傳PID數字陣列(含身分相符之根，由子孫往根排序)
 * @example
 *
 * import { findProcessDescendants } from 'w-package-tools-e2e/src/runShardedE2e.mjs'
 *
 * let table = [{ pid: 10, ppid: 1, startMs: 100 }, { pid: 11, ppid: 10, startMs: 110 }]
 * console.log(findProcessDescendants(table, [{ pid: 10, startMs: 100 }]))
 * // => [11, 10]
 *
 */
export function findProcessDescendants(table, roots, opt = {}) {
    let keys = new Set(roots.map(keyOf))
    let seeds = table.filter((row) => keys.has(keyOf(row))).map((row) => ({ row, basis: BASIS_IDENTITY }))
    return walkLiveTree(table, seeds, new Set(opt.protect || [])).map((v) => v.row.pid)
}


/**
 * 取樣：依行程表更新已確認身分之行程清單
 *
 * 只保留仍存活且身分相符者(已結束者之紀錄日後不可能再相符)，並加入沿存活父子鏈走到之新子孫；父已結束者一律不納入，避免把誤認寫進清單
 *
 * @param {Array} table 輸入行程表陣列，元素為{pid,ppid,startMs,image}
 * @param {Array} known 輸入已確認身分之行程陣列，元素為{pid,startMs,image}
 * @returns {Object} 回傳物件{known,added,dropped}，known為更新後之陣列，added、dropped為新增與移除筆數
 * @example
 *
 * import { sampleShardKnown } from 'w-package-tools-e2e/src/runShardedE2e.mjs'
 *
 * let table = [{ pid: 10, ppid: 1, startMs: 100, image: 'a' }, { pid: 11, ppid: 10, startMs: 110, image: 'b' }]
 * console.log(sampleShardKnown(table, [{ pid: 10, startMs: 100, image: 'a' }]).added)
 * // => 1
 *
 */
export function sampleShardKnown(table, known) {
    let keys = new Set(known.map(keyOf))
    let seeds = table.filter((row) => keys.has(keyOf(row))).map((row) => ({ row, basis: BASIS_IDENTITY }))
    let next = walkLiveTree(table, seeds).map((v) => ({ pid: v.row.pid, startMs: v.row.startMs, image: v.row.image || '' }))
    let nextKeys = new Set(next.map(keyOf))
    return {
        known: next,
        added: next.filter((k) => !keys.has(keyOf(k))).length,
        dropped: [...keys].filter((k) => !nextKeys.has(k)).length,
    }
}


//本行程與祖先鏈: 祖先須起始不晚於其子, 否則為 PID 回收之假關係而停止往上(停在父已結束處)
function protectedPids(table, selfPid, parentPid) {
    let set = new Set([selfPid, parentPid].filter((v) => Number.isInteger(v)))
    let byPid = new Map(table.map((p) => [p.pid, p]))
    let cur = byPid.get(selfPid)
    while (cur) {
        let par = byPid.get(cur.ppid)
        if (!par || par.pid === cur.pid || par.startMs > cur.startMs) {
            break
        }
        set.add(par.pid)
        cur = par
    }
    return set
}

//Linux: 讀 <procRoot>/<pid>/environ 與 cwd; 無此目錄之平台回 null(無證據可用)
function readEvidenceProc(pids, marker, dirs, procRoot) {
    if (!fs.existsSync(procRoot)) {
        return null
    }
    let res = new Map()
    for (let pid of pids) {
        let env = false
        let cwd = false
        try {
            env = fs.readFileSync(path.join(procRoot, String(pid), 'environ'), 'utf8').split('\0').includes(marker)
        }
        catch (err) {}
        try {
            let c = fs.readlinkSync(path.join(procRoot, String(pid), 'cwd'))
            let p = c.endsWith(path.sep) ? c : `${c}${path.sep}`
            cwd = dirs.some((d) => p.startsWith(d))
        }
        catch (err) {}
        res.set(pid, { env, cwd })
    }
    return res
}

 * @param {String} opt.marker 輸入環境變數標記字串，格式為'E2E_SHARD_RUN=<id>'
 * @param {Array} opt.dirs 輸入目錄字串陣列，每個須以路徑分隔符結尾
 * @param {Function} [opt.exec] 輸入同步執行子指令函數(與child_process.spawnSync同形)，src為'evidence'，另帶targets(PID陣列)，預設為不經shell、隱藏視窗之spawnSync
 * @param {String} [opt.platform=process.platform] 輸入作業系統字串，預設process.platform
 * @param {String} [opt.procRoot='/proc'] 輸入非Windows平台之行程資訊目錄字串，預設'/proc'
 * @returns {Map|null} 回傳Map，鍵為PID，值為{env,cwd}布林值物件；整批讀取失敗或平台無法讀取回傳null
 * @example
 *
 * import { readProcessEvidence } from 'w-package-tools-e2e/src/runShardedE2e.mjs'
 *
 * let m = readProcessEvidence([1234], { marker: 'E2E_SHARD_RUN=abc', dirs: ['C:\\p\\tmp\\e2e-shards\\'] })
 * console.log(m.get(1234))
 * // => { env: true, cwd: true }
 *
 */
export function readProcessEvidence(pids, opt = {}) {
    let { marker, dirs = [], exec, platform = process.platform, procRoot = '/proc' } = opt
    if (platform !== 'win32') {
        return readEvidenceProc(pids, marker, dirs, procRoot)
    }
    let b64 = (v) => `[Text.Encoding]::UTF8.GetString([Convert]::FromBase64String('${Buffer.from(v).toString('base64')}'))`
    let script = [
        '$ErrorActionPreference=\'Stop\'',
        'Add-Type -TypeDefinition @\'',
        CS_EVIDENCE,
        '\'@',
        `$mark=${b64(marker)}`,
        `$dirs=[string[]]@(${dirs.map(b64).join(',')})`,
        `foreach ($id in @(${pids.join(',')})) { "$id " + [ShardEv]::Check($id, $mark, $dirs) }`,
    ].join('\n')
    let r
    try {
        r = (exec || execDef)('powershell', psArgs(script), { src: 'evidence', targets: [...pids], timeout: 60000, maxBuffer: 1024 * 1024 })
    }
    catch (err) {
        return null
    }
    if (!r || r.error || r.status !== 0) {
        return null
    }
    let res = new Map()
    for (let l of String(r.stdout || '').split(/\r?\n/)) {
        let m = /^(\d+) env=([01]) cwd=([01])$/.exec(l.trim())
        if (m) {
            res.set(Number(m[1]), { env: m[2] === '1', cwd: m[3] === '1' })
        }
    }
    return res
}


//下殺已確認身分之行程(不帶 /T: 子孫已自行走訪、逐筆留紀錄)。Windows 先取 handle 釘住行程再比對起始時間, 比對與下殺之間 PID 不可能被回收。回傳 Map(pid → 結果)
function killIdentified(rows, ctx = {}) {
    let { exec, platform = process.platform } = ctx
    let run = exec || execDef
    let targets = rows.map((r) => ({ pid: r.pid, startMs: r.startMs }))
    let res = new Map()
    if (platform === 'win32') {
        let list = rows.map((r) => `'${r.pid}|${new Date(r.startMs).toISOString()}'`).join(',')
        let script = [
            `foreach ($t in @(${list})) {`,
            '  $a = $t.Split(\'|\'); $id = [int]$a[0]; $p = $null',
            '  try { $p = [Diagnostics.Process]::GetProcessById($id) } catch { "$id gone"; continue }',
            '  try {',
            '    $null = $p.Handle',
            '    $s = $p.StartTime.ToUniversalTime().ToString(\'yyyy-MM-ddTHH:mm:ss.fffZ\', [Globalization.CultureInfo]::InvariantCulture)',
            '    if ($s -ne $a[1]) { "$id mismatch"; continue }',
            '    $p.Kill()',
            '    if ($p.WaitForExit(5000)) { "$id killed" } else { "$id pending" }',
            '  } catch { "$id error" }',
            '}',
        ].join('\n')
        let r
        try {
            r = run('powershell', psArgs(script), { src: 'kill', targets, timeout: 60000, maxBuffer: 1024 * 1024 })
        }
        catch (err) {
            r = null
        }
        if (r && !r.error && r.status === 0) {
            for (let l of String(r.stdout || '').split(/\r?\n/)) {
                let m = /^(\d+) (\w+)$/.exec(l.trim())
                if (m) {
                    res.set(Number(m[1]), m[2])
                }
            }
        }
        rows.forEach((v) => {
            if (!res.has(v.pid)) {
                res.set(v.pid, 'exec-failed')
            }
        })
        return res
    }
    for (let v of rows) {
        let r
        try {
            r = run('sh', ['-c', 'kill -9 "$1"', 'sh', String(v.pid)], { src: 'kill', targets: [{ pid: v.pid, startMs: v.startMs }], timeout: 15000, maxBuffer: 1024 * 1024 })
        }
        catch (err) {
            r = null
        }
        res.set(v.pid, r && !r.error && r.status === 0 ? 'killed' : 'exec-failed')
    }
    return res
}


//下殺紀錄之一筆
function mkKillRecord(row, basis, phase, result) {
    return {
        pid: row.pid,
        image: row.image || '',
        startMs: Number.isFinite(row.startMs) ? row.startMs : null,
        start: Number.isFinite(row.startMs) ? new Date(row.startMs).toISOString() : '',
        basis,
        phase,
        result,
    }
}


//紀錄之輸出列
function fmtKillRecord(rec) {
    return `[shard] 下殺 PID ${rec.pid} ${rec.image || '(映像名未知)'} 起始 ${rec.start || '(未知)'} 依據 ${rec.basis} 階段 ${rec.phase} 結果 ${rec.result}`
}


/**
 * 清理一次執行留下之行程：只殺身分已正面確認者，每筆留紀錄
 *
 * 可下殺者：①PID與起始時間皆與存活時之紀錄完全相同(依據identity)；②由身分已確認之存活行程沿存活父子鏈走到之子孫(依據chain:<父PID>)；
 * ③父已結束且起始不早於since(或為extra所列之監聽者)，環境變數帶本次標記(依據env-marker)或工作目錄位於實例目錄內(依據instance-cwd)者及其存活子孫。
 * 其餘一律不殺：沒有行程表、讀不到證據、舊紀錄之PID已被回收皆然。本行程與祖先鏈永不下殺，亦不經其往下走。
 * 下殺不帶/T；Windows於同一handle上比對起始時間後才終止。
 *
 * @param {Object} opt 輸入設定物件
 * @param {Array} [opt.known=[]] 輸入存活時記下之行程陣列，元素為{pid,startMs}
 * @param {String} [opt.marker=''] 輸入環境變數標記字串，空字串不查證據
 * @param {Array} [opt.dirs=[]] 輸入實例目錄字串陣列，每個須以路徑分隔符結尾
 * @param {Number} [opt.since] 輸入起始時間下限毫秒數，早於此者不視為本次之孤兒候選；未給不查孤兒
 * @param {Array} [opt.extra=[]] 輸入額外須查證據之行程陣列，元素為{pid,port}(仍占埠之監聽者)，無證據者另記警告
 * @param {String} [opt.phase=''] 輸入階段字串，寫入紀錄
 * @param {Function} [opt.exec] 輸入同步執行子指令函數(與child_process.spawnSync同形)，src為'table'、'evidence'、'kill'(後二者另帶targets)，預設為不經shell、隱藏視窗之spawnSync
 * @param {String} [opt.platform=process.platform] 輸入作業系統字串，預設process.platform
 * @param {Number} [opt.selfPid=process.pid] 輸入本行程PID，預設process.pid
 * @param {Number} [opt.parentPid=process.ppid] 輸入本行程之父PID，預設process.ppid
 * @param {String} [opt.procRoot='/proc'] 輸入非Windows平台之行程資訊目錄字串，預設'/proc'
 * @param {Function} [opt.log] 輸入輸出函數，預設不輸出
 * @returns {Object} 回傳物件{tableOk,kills,skipped}，kills元素為{pid,image,startMs,start,basis,phase,result}，skipped元素為{pid,image,startMs,start,reason,phase,port}，reason為'protected'、'no-evidence'、'evidence-unreadable'
 * @example
 *
 * import { reapShardProcesses } from 'w-package-tools-e2e/src/runShardedE2e.mjs'
 *
 * let r = reapShardProcesses({ known: [{ pid: 1234, startMs: 1793319707455 }], phase: 'teardown', log: console.log })
 * console.log(r.kills.map((v) => v.basis))
 * // => ['identity']
 *
 */
export function reapShardProcesses(opt = {}) {
    let { known = [], marker = '', dirs = [], since, extra = [], phase = '', exec, platform = process.platform, selfPid = process.pid, parentPid = process.ppid, procRoot, log = () => {} } = opt
    let ctx = { exec, platform }
    let kills = []
    let skipped = []
    let table = listProcessTable(ctx)
    if (table === null) {
        log(`[shard] 警告: 取行程表失敗(階段 ${phase}), 無法確認身分, 一律不下殺; 由回驗埠與刪實例把關`)
        return { tableOk: false, kills, skipped }
    }
    let skip = (row, reason, port) => {
        skipped.push({ ...mkKillRecord(row, '', phase, ''), reason, ...(port ? { port } : {}) })
    }
    let protect = protectedPids(table, selfPid, parentPid)
    let keys = new Set(known.map(keyOf))
    let identified = table.filter((row) => keys.has(keyOf(row)))
    for (let row of identified.filter((v) => protect.has(v.pid))) {
        skip(row, 'protected')
        log(`[shard] 警告: PID ${row.pid} ${row.image || ''} 為本行程或其祖先, 不下殺`)
    }
    let seeds = identified.map((row) => ({ row, basis: BASIS_IDENTITY }))
    let tree = walkLiveTree(table, seeds, protect)

    //尚未確認者: 父已結束之候選與仍占埠之監聽者, 須有證據才下殺
    let inTree = new Set(tree.map((v) => v.row.pid))
    let byPid = new Map(table.map((p) => [p.pid, p]))
    let isOrphan = (row) => {
        //非 Windows: 父結束後由 init(PID 1)收養
        if (platform !== 'win32' && row.ppid === 1) {
            return true
        }
        let par = byPid.get(row.ppid)
        return !par || par.startMs > row.startMs
    }
    let cands = marker && Number.isFinite(since)
        ? table.filter((row) => row.startMs >= since && !protect.has(row.pid) && !inTree.has(row.pid) && isOrphan(row))
        : []
    let portOf = new Map()
    for (let e of extra) {
        let row = byPid.get(Number(e.pid))
        if (!row || inTree.has(row.pid)) {
            continue
        }
        if (protect.has(row.pid)) {
            skip(row, 'protected', e.port)
            continue
        }
        portOf.set(row.pid, e.port)
        if (!cands.includes(row)) {
            cands.push(row)
        }
    }
    let confirmed = []
    if (cands.length > 0) {
        let ev = marker ? readProcessEvidence(cands.map((v) => v.pid), { marker, dirs, exec, platform, ...(procRoot ? { procRoot } : {}) }) : null
        if (ev === null) {
            log(`[shard] 警告: 讀取身分證據失敗(階段 ${phase}), 未確認身分之 ${cands.length} 個行程一律不下殺`)
        }
        for (let row of cands) {
            let e = ev ? ev.get(row.pid) : null
            if (e && e.env) {
                confirmed.push({ row, basis: BASIS_ENV })
            }
            else if (e && e.cwd) {
                confirmed.push({ row, basis: BASIS_CWD })
            }
            else {
                skip(row, ev === null ? 'evidence-unreadable' : 'no-evidence', portOf.get(row.pid))
                if (portOf.has(row.pid)) {
                    log(`[shard] 警告: 埠 ${portOf.get(row.pid)} 之監聽者 PID ${row.pid} ${row.image || ''} 無身分證據, 不下殺`)
                }
            }
        }
        let nOther = skipped.filter((v) => v.phase === phase && v.reason !== 'protected' && !v.port).length
        if (nOther > 0) {
            log(`[shard] 父已結束而無身分證據、未下殺之行程 ${nOther} 個(多為他人之行程; 清單見 summary.json 之 killSkipped)`)
        }
    }
    let list = confirmed.length > 0 ? walkLiveTree(table, [...seeds, ...confirmed], protect) : tree
    if (list.length === 0) {
        return { tableOk: true, kills, skipped }
    }
    let results = killIdentified(list.map((v) => v.row), ctx)
    for (let v of list) {
        let rec = mkKillRecord(v.row, v.basis, phase, results.get(v.row.pid) || 'exec-failed')
        kills.push(rec)
        log(fmtKillRecord(rec))
    }
    return { tableOk: true, kills, skipped }
}


//鎖檔之持有者是否仍在: PID 存在且起始時間完全相符(查不到起始時間或紀錄無此欄時視為仍在, 收嚴)
function lockHolderAlive(info, ctx = {}) {
    if (!Number.isInteger(info.pid) || !pidExists(info.pid)) {
        return false
    }
    let table = listProcessTable(ctx)
    let cur = table ? table.find((p) => p.pid === info.pid) : null
    if (!cur || !Number.isFinite(info.procCreateMs)) {
        return true
    }
    return cur.startMs === info.procCreateMs
}


//取得鎖: 排他建立; 持有者仍在即拋錯; 已死之鎖在接手守衛(.takeover)內移除後重建, 免兩個接手者都以為取得
function acquireLock(workDir, info, ctx = {}) {
    fs.mkdirSync(workDir, { recursive: true })
    let fp = path.join(workDir, LOCK_NAME)
    let fpGuard = `${fp}.takeover`
    let write = () => fs.writeFileSync(fp, JSON.stringify(info), { flag: 'wx' })
    try {
        write()
        return { fp, stale: null }
    }
    catch (err) {
        if (err.code !== 'EEXIST') {
            throw err
        }
    }
    let read = () => {
        try {
            return JSON.parse(fs.readFileSync(fp, 'utf8'))
        }
        catch (err) {
            return null
        }
    }
    let judge = () => {
        let cur = read()
        if (cur) {
            return { cur, alive: lockHolderAlive(cur, ctx) }
        }
        //內容不可讀: 剛建立尚未寫完者視為仍在; 放久了才算已死
        let st = fs.statSync(fp, { throwIfNoEntry: false })
        return { cur: null, alive: !!st && Date.now() - st.mtimeMs < 60000 }
    }
    let first = judge()
    if (first.alive) {
        throw new Error(`runShardedE2e: 同一專案已有執行者(鎖 ${fp}${first.cur ? `, PID ${first.cur.pid}` : ''}), 不可並行`)
    }
    let stGuard = fs.statSync(fpGuard, { throwIfNoEntry: false })
    if (stGuard && Date.now() - stGuard.mtimeMs > 60000) {
        fs.rmSync(fpGuard, { force: true })
    }
    try {
        fs.writeFileSync(fpGuard, String(process.pid), { flag: 'wx' })
    }
    catch (err) {
        throw new Error(`runShardedE2e: 另一執行者正在接手已死之鎖(${fpGuard}), 不可並行`)
    }
    try {
        let second = judge()
        if (second.alive) {
            throw new Error(`runShardedE2e: 同一專案已有執行者(鎖 ${fp}), 不可並行`)
        }
        fs.rmSync(fp, { force: true })
        write()
        return { fp, stale: second.cur || {} }
    }
    finally {
        fs.rmSync(fpGuard, { force: true })
    }
}


//釋放鎖並刪空工作目錄(非空表示有實例未移除, 保留)
function releaseLock(lock, workDir) {
    fs.rmSync(lock.fp, { force: true })
    try {
        fs.rmdirSync(workDir)
    }
    catch (err) {}
}


//正規化排除路徑
function normExclude(list) {
    return list.map((v) => {
        let s = String(v).replace(/\\/g, '/').replace(/^\.\//, '').replace(/\/+$/, '')
        if (s === '' || s === '.' || path.isAbsolute(s) || s.split('/').includes('..')) {
            throw new Error(`runShardedE2e: exclude 須為專案根內之相對路徑, 實得「${v}」`)
        }
        return s
    })
}


//排除判斷函數: rel 等於排除項或位於其下即排除(Windows 不分大小寫)
function makeExcluded(list) {
    let fold = process.platform === 'win32' ? (s) => s.toLowerCase() : (s) => s
    let items = list.map(fold)
    return (rel) => {
        let r = fold(rel)
        return items.some((e) => r === e || r.startsWith(`${e}/`))
    }
}


//複製專案樹(只用 OS 複製, 不讀檔案內容; .env 亦然); 遇連結即拋錯(複製前已檢查, 此處防中途出現)
async function copyTree(root, dst, isExcluded) {
    let walk = async (dir, dstDir) => {
        await fs.promises.mkdir(dstDir, { recursive: true })
        for (let ent of await fs.promises.readdir(dir, { withFileTypes: true })) {
            let p = path.join(dir, ent.name)
            let rel = path.relative(root, p).replace(/\\/g, '/')
            if (isExcluded(rel)) {
                continue
            }
            let q = path.join(dstDir, ent.name)
            if (ent.isSymbolicLink()) {
                throw new Error(`runShardedE2e: 複製途中遇到連結: ${p}`)
            }
            if (ent.isDirectory()) {
                await walk(p, q)
            }
            else if (ent.isFile()) {
                await fs.promises.copyFile(p, q, fs.constants.COPYFILE_EXCL)
            }
        }
    }
    await walk(root, dst)
}


//專案是否已依 E2E_PORT_OFFSET 接線: 未接線時各組都用基準埠, 依沿用政策互打而不報錯(只掃 test/ 與 test/tools/ 之 .mjs)
function isPortOffsetAdopted(root, testDir) {
    let fds = [path.join(root, testDir), path.join(root, testDir, 'tools')]
    for (let fd of fds) {
        if (!fs.existsSync(fd)) {
            continue
        }
        for (let f of fs.readdirSync(fd)) {
            if (f.endsWith('.mjs') && /E2E_PORT_OFFSET|resolveE2ePort/.test(fs.readFileSync(path.join(fd, f), 'utf8'))) {
                return true
            }
        }
    }
    return false
}


//時間字串(輸出夾名)
function stamp() {
    let d = new Date()
    let z = (n, w = 2) => String(n).padStart(w, '0')
    return `${d.getFullYear()}${z(d.getMonth() + 1)}${z(d.getDate())}-${z(d.getHours())}${z(d.getMinutes())}${z(d.getSeconds())}-${z(d.getMilliseconds(), 3)}`
}

}

    if (plan.unknownFiles.length > 0) {

    }
}


/**
 * 把一個專案之e2e檔分組，各組放進以埠位移之臨時實例同時跑，彙整結果後移除實例(只跑比對，不寫標準圖)
 *
 * 流程：讀設定(test/tools/e2eShard.json，可不存在，opt可覆寫)→列舉檔案與分組(含記憶體守門)→取鎖→清除上次中斷之殘留→預檢各實例之每個埠可綁定(不沿用；未過者於建立實例前拋錯並列出占用者)→
 * 檢查來源樹無額外連結→建立實例(排除清單以外整份複製；node_modules與test/pics以junction指回；.env等只複製不讀取)→各實例以E2E_PORT_OFFSET=<組序×portStep>執行mocha(cwd為實例根，輸出分寫各組log，不彈視窗)→彙整→收尾。
 * 收尾(成功、失敗、例外、SIGINT、SIGTERM、SIGBREAK、SIGHUP皆執行；Windows之taskkill /F與child.kill為TerminateProcess，處理器來不及執行，由下次開跑之殘留清除處理)：樹狀終止子行程與孤兒→回驗本次各埠已釋放→搬回各實例testPending(不覆蓋)→依全域技能full-run-and-instances §5.5拆連結、回驗、遞迴刪除並確認已不存在。任一步失敗大聲報錯、不繼續刪除該實例、離開碼非0。
 * 未經正面確認身分之行程一律不殺(reapShardProcesses)：取樣、收尾與下次開跑之殘留清除皆同；每次下殺之PID、映像名、起始時間與判定依據寫入[shard]輸出與summary.json。
 * log於執行中即直接寫入outDir(行程被硬殺仍保留)。有失敗、未跑完、某組沒有統計或收尾失敗，exitCode即非0；收到訊號時為128加訊號值。
 * 專案須已依E2E_PORT_OFFSET宣告埠(README §2.10)，ports未給、E2E_REGEN=1或mochaArgs含--baseline皆拋錯。
 *
 * @param {Object} opt 輸入設定物件
 * @param {String} opt.root 輸入專案根目錄字串
 * @param {String} [opt.configPath='test/tools/e2eShard.json'] 輸入設定檔路徑字串，相對路徑以root解析，檔案可不存在
 * @param {Array} [opt.ports] 輸入基準埠陣列，覆寫設定檔；實跑時必須給(設定檔或此處)，專案確實無服務者明寫[]
 * @param {Number} [opt.instances=5] 輸入實例數，覆寫設定檔，預設5
 * @param {Array} [opt.officialOnly=[]] 輸入只能跑在位移0之檔名陣列(畫面或期望值含主機、埠者)，覆寫設定檔，預設[]
 * @param {Array} [opt.together=[]] 輸入必須同組之檔名陣列之陣列，覆寫設定檔，預設[]
 * @param {Object} [opt.durations={}] 輸入各檔耗時(分)物件，只用於分組，覆寫設定檔，缺者以已知檔之平均值估算，預設{}
 * @param {Array} [opt.exclude=[]] 輸入額外不複製之路徑陣列(相對root)，覆寫設定檔，預設[]
 * @param {Number} [opt.portStep=100] 輸入各組之埠位移間距，覆寫設定檔，預設100
 * @param {Number} [opt.memPerInstanceGb=1.5] 輸入每實例需要之記憶體(GB)，預設1.5
 * @param {Number} [opt.availMemGb] 輸入可用記憶體(GB)，預設os.freemem()，供測試注入
 * @param {String} [opt.testDir='test'] 輸入測試檔目錄字串(相對root)，預設'test'
 * @param {String} [opt.workDir='<root>/tmp/e2e-shards'] 輸入實例與鎖檔之工作目錄字串，相對路徑以root解析，實例位於<workDir>/inst<N>/<專案名>；實例含.env等憑證副本，須位於gitignore範圍內
 * @param {String} [opt.outDir='<root>/tmp/e2e-shards-out/<時間>'] 輸入各組log、統計與summary.json之輸出目錄字串，相對路徑以root解析
 * @param {Array} [opt.mochaArgs=['--timeout','180000']] 輸入附加於檔案之後之mocha參數陣列，不可含--reporter、-R、--baseline、--parallel、-p
 * @param {Number} [opt.groupTimeoutMs=14400000] 輸入每組之執行上限毫秒數，逾時樹狀終止該組並判為未跑完，預設14400000(4小時)
 * @param {Number} [opt.progressMs=300000] 輸入進度列印間隔毫秒數，預設300000(5分)
 * @param {Number} [opt.treeSampleMs=30000] 輸入取樣各組子孫行程之間隔毫秒數，預設30000；取樣結果(只含身分已確認者)寫入鎖檔，供父子鏈中斷時(中間行程已結束)仍能以PID與起始時間辨識孤兒
 * @param {Function} [opt.exec] 輸入同步執行子指令函數，供測試注入行程表與下殺，格式為(prog, args, { src, timeout, maxBuffer })=>{ status, stdout, error }(與child_process.spawnSync同形)，src為'table'、'evidence'、'kill'(後二者另帶targets)、'tasklist'及listenerPids之'netstat'、'powershell'、'lsof'，預設為不經shell、隱藏視窗之spawnSync
 * @param {Boolean} [opt.plan=false] 輸入是否只印分組與預估布林值，為true時不取鎖、不建立任何東西、不執行，預設false
 * @param {Boolean} [opt.updateDurations=false] 輸入是否把實測耗時寫回設定檔布林值，預設false(只印不改檔)
 * @param {Boolean} [opt.debug=false] 輸入是否印出因非預期值而改用預設之選項布林值，預設false
 * @param {Function} [opt.log=console.log] 輸入輸出函數，預設console.log
 * @param {Object} [opt.proc=process] 輸入註冊訊號處理器之對象，供測試注入，預設process
 * @returns {Promise} 回傳Promise，resolve回傳物件{planned,exitCode,signal,plan,groups,totals,wallMs,durations,outDir,teardownErrors,kills,killSkipped,knownCount}，groups元素為{index,offset,ports,files,ms,exit,signal,passing,failing,pending,total,complete,note,failures,logPath,resultPath}，totals為{passing,failing,pending,failures,incomplete}，kills元素為{pid,image,startMs,start,basis,phase,result}，killSkipped元素為{pid,image,startMs,start,reason,phase,port}，knownCount為{current,total}；預檢、設定、鎖等開跑前之問題reject(err.kills帶已發生之下殺紀錄，有紀錄時亦寫入outDir之summary.json)
 * @example
 *
 * import runShardedE2e from 'w-package-tools-e2e/src/runShardedE2e.mjs'
 *
 * let r = await runShardedE2e({ root: process.cwd(), ports: [11006, 8090], instances: 5 })
 * process.exit(r.exitCode)
 *
 */
async function runShardedE2e(opt = {}) {
    if (!isObj(opt) || typeof opt.root !== 'string' || opt.root === '') {
        throw new Error('runShardedE2e: root 為必填字串')
    }
    let root = path.resolve(opt.root)
    if (!fs.existsSync(root)) {
        throw new Error(`runShardedE2e: root 不存在: ${root}`)
    }
    let log = typeof opt.log === 'function' ? opt.log : console.log
    let proc = opt.proc && typeof opt.proc.on === 'function' ? opt.proc : process
    let ctx = { exec: typeof opt.exec === 'function' ? opt.exec : undefined }
    let probs = []

    //設定: opt 覆寫設定檔
    let fpConfig = path.resolve(root, typeof opt.configPath === 'string' && opt.configPath ? opt.configPath : path.join('test', 'tools', 'e2eShard.json'))
    let { cfg } = readShardConfig(fpConfig)
    let pick = (k) => (opt[k] !== undefined && opt[k] !== null ? opt[k] : cfg[k])
    let ports = pick('ports')
    let portsGiven = ports !== undefined && ports !== null
    ports = portsGiven ? checkPorts(ports) : []
    let portStep = pickPosInt(pick('portStep'), 100, 'portStep', probs)
    let memPerInstanceGb = pickPos(opt.memPerInstanceGb, 1.5, 'memPerInstanceGb', probs)
    let testDir = typeof opt.testDir === 'string' && opt.testDir ? opt.testDir : 'test'
    let groupTimeoutMs = pickPosInt(opt.groupTimeoutMs, 14400000, 'groupTimeoutMs', probs)
    let progressMs = pickPosInt(opt.progressMs, 300000, 'progressMs', probs)
    let treeSampleMs = pickPosInt(opt.treeSampleMs, 30000, 'treeSampleMs', probs)
    let mochaArgs = opt.mochaArgs
    if (!Array.isArray(mochaArgs) || mochaArgs.some((v) => typeof v !== 'string')) {
        if (mochaArgs !== undefined && mochaArgs !== null) {
            probs.push('mochaArgs 須為字串陣列, 改用預設')
        }
        mochaArgs = ['--timeout', '180000']
    }
    let exUser = pick('exclude')
    if (exUser !== undefined && exUser !== null && !Array.isArray(exUser)) {
        probs.push('exclude 須為陣列, 改用預設 []')
        exUser = []
    }
    let exclude = normExclude([...DEFAULT_EXCLUDE, ...(exUser || [])])
    let workDir = path.resolve(root, typeof opt.workDir === 'string' && opt.workDir ? opt.workDir : path.join('tmp', 'e2e-shards'))
    let outDir = path.resolve(root, typeof opt.outDir === 'string' && opt.outDir ? opt.outDir : path.join('tmp', 'e2e-shards-out', stamp()))
    if (opt.debug === true) {
        probs.forEach((p) => log(`[shard] 選項非預期: ${p}`))
    }

    //工作目錄不得為專案根或其上層(刪除範圍), 也不得落在連結目標內
    let relRootFromWork = path.relative(workDir, root)
    if (relRootFromWork === '' || (!relRootFromWork.startsWith('..') && !path.isAbsolute(relRootFromWork))) {
        throw new Error(`runShardedE2e: workDir 不可為專案根或其上層: ${workDir}`)
    }
    if (makeExcluded(['node_modules', 'test/pics', '.git'])(path.relative(root, workDir).replace(/\\/g, '/'))) {
        throw new Error(`runShardedE2e: workDir 不可位於 node_modules、test/pics、.git 內: ${workDir}`)
    }
    //工作目錄與輸出目錄在專案樹內時不複製
    let inRoot = (p) => {
        let r = path.relative(root, p)
        return r !== '' && !r.startsWith('..') && !path.isAbsolute(r) ? r.replace(/\\/g, '/') : ''
    }
    let isExcluded = makeExcluded([...exclude, inRoot(workDir), inRoot(outDir)].filter(Boolean))

    //列舉與分組
    let fdTest = path.resolve(root, testDir)
    let files = fs.existsSync(fdTest) ? fs.readdirSync(fdTest).filter((f) => RE_E2E.test(f)).sort() : []
    if (files.length === 0) {
        throw new Error(`runShardedE2e: ${fdTest} 內沒有 e2e-*.test.mjs`)
    }
    let availMemGb = typeof opt.availMemGb === 'number' ? opt.availMemGb : os.freemem() / 2 ** 30
    let plan = planE2eShards({
        files,
        instances: pick('instances'),
        officialOnly: pick('officialOnly'),
        together: pick('together'),
        durations: pick('durations'),
        availMemGb,
        memPerInstanceGb,
    })
    let table = buildPortTable(plan.groups, ports, portStep)
    printPlan(plan, table, portStep, log)
    let base = { plan, groups: [], totals: { passing: 0, failing: 0, pending: 0, failures: [], incomplete: [] }, wallMs: 0, durations: {}, outDir, teardownErrors: [], kills: [], killSkipped: [], knownCount: { current: 0, total: 0 } }

    if (opt.plan === true) {
        if (!portsGiven) {
            log('[shard] 警告: 未給 ports, 實跑時會拋錯(專案確實無服務者明寫 ports: [])')
        }
        return { planned: true, exitCode: 0, signal: null, ...base }
    }

    //開跑前之守門: 皆不建立任何東西
    if (!portsGiven) {
        throw new Error('runShardedE2e: 未給 ports, 無法預檢(不預檢即可能打到別人的服務而不報錯); 專案確實無服務者明寫 ports: []')
    }
    if (process.env.E2E_REGEN === '1') {
        throw new Error('runShardedE2e: 本工具只跑比對; 環境變數 E2E_REGEN=1 會經 test/pics 之 junction 寫入正式標準圖, 請先移除')
    }
    let badArgs = mochaArgs.filter((v) => ['--baseline', '--reporter', '-R', '--parallel', '-p'].includes(v) || /^--reporter=/.test(v))
    if (badArgs.length > 0) {
        throw new Error(`runShardedE2e: mochaArgs 不可含 ${badArgs.join(', ')}(--baseline 會寫入正式標準圖; 報告器由本工具提供; 固定 --no-parallel)`)
    }
    let hasNm = fs.existsSync(path.join(root, 'node_modules', 'mocha', 'bin', 'mocha.js'))
    if (!hasNm) {
        throw new Error(`runShardedE2e: 找不到 ${path.join(root, 'node_modules', 'mocha', 'bin', 'mocha.js')}`)
    }
    if (ports.length > 0 && plan.groups.length > 1 && !isPortOffsetAdopted(root, testDir)) {
        throw new Error(`runShardedE2e: ${testDir}/ 與 ${testDir}/tools/ 之 .mjs 皆未使用 resolveE2ePort 或 E2E_PORT_OFFSET; 專案未接線時各組都用基準埠, 依沿用政策互打而不報錯(導入方式見 README §2.10)`)
    }

    let projName = path.basename(root)
    let instOf = (i) => path.join(workDir, `inst${i}`)
    let projOf = (i) => path.join(instOf(i), projName)
    log(`[shard] 專案根 ${root}`)
    log(`[shard] 工作目錄 ${workDir}`)
    log(`[shard] 輸出目錄 ${outDir}`)

    //連結目標: node_modules 只看兩層名稱(執行中可能新增快取), test/pics 整棵逐檔(工具只讀比對, 不得有任何變動); expectPics 為執行前之快照
    let targetsOf = (expectPics) => [
        { path: path.join(root, 'node_modules'), depth: 2, meta: false, allowAdded: true },
        ...(fs.existsSync(path.join(root, 'test', 'pics')) ? [{ path: path.join(root, 'test', 'pics'), ...(expectPics ? { expect: expectPics } : {}) }] : []),
    ]

    //鎖檔: procCreateMs 為行程表之真實起始時間(取不到時為 null, 他人視為仍在); runId 為本次之環境變數標記; known 只含身分已確認者
    let runId = randomUUID()
    let tableSelf = listProcessTable(ctx)
    let rowSelf = tableSelf ? tableSelf.find((p) => p.pid === process.pid) : null
    let lockInfo = { pid: process.pid, procCreateMs: rowSelf ? rowSelf.startMs : null, startMs: Date.now(), root, runId, known: [] }
    let lock = acquireLock(workDir, lockInfo, ctx)
    let writeLock = () => fs.writeFileSync(lock.fp, JSON.stringify(lockInfo))

    let t0 = Date.now()
    let kills = []
    let killSkipped = []
    let knownEver = new Set()
    let dirs = [`${workDir}${path.sep}`]
    //下殺一律經此: 只殺身分已確認者並留紀錄
    let reap = (o) => {
        let r = reapShardProcesses({ ...ctx, dirs, log, ...o })
        kills.push(...r.kills)
        killSkipped.push(...r.skipped)
        return r
    }
    let signal = null
    let groups = []
    let started = []
    let created = []
    let onAbort = []
    let fatal = null
    let teardownErrors = []
    let picsBefore = null
    let timerProgress = null
    let timerSample = null

    let handlers = SIGNALS.map((sig) => {
        let h = () => {
            if (signal) {
                log(`[shard] 再次收到 ${sig}: 收尾中, 請稍候(中斷收尾會留下實例)`)
                return
            }
            signal = sig
            log(`[shard] 收到 ${sig}: 終止子行程並收尾`)
            onAbort.forEach((f) => f())
        }
        proc.on(sig, h)
        return [sig, h]
    })

    //清除一個實例(搬回證據、拆連結、回驗、刪除); 失敗只記錄, 不影響其他實例之清除
    let removeInst = (instRoot, expectPics) => {
        let pendingMoves = []
        if (fs.existsSync(instRoot)) {
            for (let ent of fs.readdirSync(instRoot, { withFileTypes: true })) {
                if (ent.isDirectory()) {
                    pendingMoves.push({ from: path.join(instRoot, ent.name, 'testPending'), to: path.join(root, 'testPending') })
                }
            }
        }
        removeE2eInstance({ instRoot, workDir, pendingMoves, targets: targetsOf(expectPics), log })
    }

    try {
        //上一次中斷之殘留: 先殺其存活且身分相符之子行程, 再依拆除程序清掉; 鎖檔可能已存放數日, 同樣逐筆比對
        if (lock.stale) {
            if (typeof lock.stale.runId === 'string' && lock.stale.runId !== '') {
                log(`[shard] 接手已死之鎖(PID ${lock.stale.pid}): 終止其殘留之子行程`)
                let r = reap({
                    known: Array.isArray(lock.stale.known) ? lock.stale.known : [],
                    marker: `${MARK_ENV}=${lock.stale.runId}`,
                    since: lock.stale.startMs,
                    phase: 'stale-lock',
                })
                log(`[shard] 上次殘留: 鎖檔 known ${Array.isArray(lock.stale.known) ? lock.stale.known.length : 0} 筆, 下殺 ${r.kills.length} 筆`)
            }
            else if (Array.isArray(lock.stale.groups) && lock.stale.groups.length > 0) {
                log(`[shard] 警告: 接手已死之舊格式鎖檔(PID ${lock.stale.pid}), 其 ${lock.stale.groups.length} 筆紀錄無法確認身分, 不下殺; 如有殘留由預檢與刪實例報錯`)
            }
        }
        let staleInsts = fs.readdirSync(workDir).filter((d) => /^inst\d+$/.test(d))
        for (let d of staleInsts) {
            log(`[shard] 清除上次中斷之殘留實例 ${path.join(workDir, d)}`)
            removeInst(path.join(workDir, d), null)
        }
        //剛殺掉殘留行程時埠可能還要片刻才釋放; 真被占用者下方預檢照樣擋下
        if (lock.stale) {
            await waitPortsFree(table.map((t) => t.port), 10000)
        }

        //預檢: 每個實例之每個埠須能綁定; 未過即拋錯, 不沿用, 不建立實例
        let blocked = []
        for (let t of table) {
            let r = await checkPortFree(t.port)
            if (!r.free) {
                blocked.push({ ...t, host: r.host, code: r.code, owners: describeOwners(t.port, ctx) })
            }
        }
        if (blocked.length > 0) {
            let err = new Error(`runShardedE2e: 預檢失敗, 下列埠無法綁定(不沿用, 未建立任何實例):\n${blocked.map((b) => `  第 ${b.group} 組 埠 ${b.port}(基準 ${b.base}+${b.offset}): ${b.code}@${b.host}; ${b.owners}`).join('\n')}`)
            err.blocked = blocked
            throw err
        }

        //來源樹內之連結: 不複製之路徑以外不得有連結(node_modules、test/pics 屬排除項, 不在此列)
        let extra = listDirLinks(root, { skip: (rel) => isExcluded(rel) })
        if (extra.length > 0) {
            throw new Error(`runShardedE2e: 來源樹內有不允許之連結(未建立任何實例; 請改為實體檔或列入 exclude):\n${extra.map((l) => `  ${l.rel} -> ${l.target}`).join('\n')}`)
        }

        fs.mkdirSync(outDir, { recursive: true })
        picsBefore = fs.existsSync(path.join(root, 'test', 'pics')) ? snapshotLinkTarget(path.join(root, 'test', 'pics')) : null

        //建立實例
        for (let g of plan.groups) {
            if (signal) {
                break
            }
            let dst = projOf(g.index)
            log(`[shard] 建立實例 ${dst}`)
            created.push(g.index)
            await copyTree(root, dst, isExcluded)
            fs.symlinkSync(path.join(root, 'node_modules'), path.join(dst, 'node_modules'), 'junction')
            if (fs.existsSync(path.join(root, 'test', 'pics'))) {
                fs.mkdirSync(path.join(dst, 'test'), { recursive: true })
                fs.symlinkSync(path.join(root, 'test', 'pics'), path.join(dst, 'test', 'pics'), 'junction')
            }
            //logs 被排除, 但服務啟動時常需此夾
            fs.mkdirSync(path.join(dst, 'logs'), { recursive: true })
            let links = listDirLinks(instOf(g.index)).map((l) => l.rel)
            let expectLinks = [`${projName}/node_modules`, ...(fs.existsSync(path.join(root, 'test', 'pics')) ? [`${projName}/test/pics`] : [])].sort()
            if (JSON.stringify(links) !== JSON.stringify(expectLinks)) {
                throw new Error(`runShardedE2e: 實例內之連結不符預期: ${links.join(', ')}`)
            }
            let lack = g.files.filter((f) => !fs.existsSync(path.join(dst, testDir, f)))
            if (lack.length > 0) {
                throw new Error(`runShardedE2e: 實例內缺測試檔(exclude 排除了?): ${lack.join(', ')}`)
            }
        }

        //根被樹狀終止前留紀錄(taskkill /T 不回報殺了誰, 故以當下之行程表列出; 取不到時只記根)
        let recordStop = (run) => {
            if (run.stopRecorded) {
                return
            }
            run.stopRecorded = true
            let tb = listProcessTable(ctx)
            let row = tb ? tb.find((p) => p.pid === run.pid) : null
            let list = row ? walkLiveTree(tb, [{ row, basis: BASIS_OWN }]) : [{ row: { pid: run.pid, image: '', startMs: null }, basis: BASIS_OWN }]
            for (let v of list) {
                let rec = mkKillRecord(v.row, v.basis, 'stop', 'tree-kill-requested')
                kills.push(rec)
                log(fmtKillRecord(rec))
            }
        }

        //執行
        let runGroup = (g) => {
            let offset = g.index * portStep
            let cwd = projOf(g.index)
            let logPath = path.join(outDir, `group${g.index}.log`)
            let resultPath = path.join(outDir, `group${g.index}.result.json`)
            let rec = {
                index: g.index,
                offset,
                ports: table.filter((t) => t.group === g.index).map((t) => t.port),
                files: g.files,
                ms: 0,
                exit: null,
                signal: null,
                passing: 0,
                failing: 0,
                pending: 0,
                total: 0,
                complete: false,
                note: '',
                failures: [],
                logPath,
                resultPath,
                fileMs: {},
            }
            groups.push(rec)
            let fd = fs.openSync(logPath, 'a')
            let args = [
                'node_modules/mocha/bin/mocha.js',
                '--no-parallel',
                ...g.files.map((f) => `${testDir}/${f}`.replace(/\\/g, '/')),
                ...mochaArgs,
                '--reporter',
                REPORTER,
            ]
            let tStart = Date.now()
            let child = spawn(process.execPath, args, {
                cwd,
                env: { ...process.env, E2E_PORT_OFFSET: String(offset), E2E_SHARD_RESULT: resultPath, [MARK_ENV]: runId },
                stdio: ['ignore', fd, fd],
                windowsHide: true,
                detached: process.platform !== 'win32',
            })
            fs.closeSync(fd)
            let run = { rec, child, pid: child.pid, startMs: tStart, timedOut: false, done: false, rootSeen: false, stopRecorded: false, known: [] }
            started.push(run)
            run.promise = new Promise((resolve) => {
                let finish = (code, sig, spawnErr) => {
                    if (run.done) {
                        return
                    }
                    run.done = true
                    rec.ms = Date.now() - tStart
                    rec.exit = code
                    rec.signal = sig
                    if (spawnErr) {
                        rec.note = `無法啟動 mocha(${spawnErr.message})`
                    }
                    resolve()
                }
                run.finish = finish
                child.once('error', (err) => finish(null, null, err))
                child.once('close', (code, sig) => finish(code, sig, null))
            })
            run.stop = () => {
                if (!run.done && isChildAlive(child)) {
                    recordStop(run)
                    try {
                        killOwnTree(child)
                    }
                    catch (err) {}
                }
                //kill 後 close 遲遲不來時不得卡住收尾
                if (!run.tForce) {
                    run.tForce = setTimeout(() => run.finish(null, null, null), 20000)
                }
            }
            let tTimeout = setTimeout(() => {
                run.timedOut = true
                log(`[shard] 第 ${g.index} 組逾時(${min2(groupTimeoutMs)} 分), 終止`)
                run.stop()
            }, groupTimeoutMs)
            tTimeout.unref()
            run.promise.then(() => {
                clearTimeout(tTimeout)
                clearTimeout(run.tForce)
            })
            return run
        }

        if (!signal) {
            onAbort.push(() => started.forEach((r) => r.stop()))
            plan.groups.forEach(runGroup)
            let readStat = (p) => {
                try {
                    return JSON.parse(fs.readFileSync(p, 'utf8'))
                }
                catch (err) {
                    return null
                }
            }
            timerProgress = setInterval(() => {
                let parts = started.map((r) => {
                    let s = readStat(r.rec.resultPath)
                    let cnt = s ? `${s.passing + s.failing + s.pending}/${s.total}(✘${s.failing})` : '尚無統計'
                    return `第${r.rec.index}組 ${r.done ? '完成' : '執行中'} ${cnt}`
                })
                log(`[shard] 進度 ${min2(Date.now() - t0)} 分: ${parts.join(' | ')}`)
            }, progressMs)
            //取樣子孫: 根或中間行程被硬殺後父子鏈即斷(Windows 之 libuv job 只連帶殺直屬子行程, 其下如 shell:true 之服務成孤兒), 故趁存活時記下 PID 與起始時間
            //根: 持有 child 且尚未 exit 時 PID 不會被回收, 由行程表取真實起始時間; 其餘只沿存活父子鏈走, 誤認不會進 known
            let sampleNow = () => {
                let tb = listProcessTable(ctx)
                if (!tb) {
                    return
                }
                let changed = false
                for (let r of started) {
                    let seed = r.known
                    if (!r.rootSeen && !r.done && isChildAlive(r.child)) {
                        let row = tb.find((p) => p.pid === r.pid)
                        if (row) {
                            r.rootSeen = true
                            changed = true
                            seed = [...r.known, { pid: row.pid, startMs: row.startMs, image: row.image || '' }]
                        }
                    }
                    let s = sampleShardKnown(tb, seed)
                    r.known = s.known
                    r.known.forEach((k) => knownEver.add(`${k.pid}@${k.startMs}`))
                    changed = changed || s.added > 0 || s.dropped > 0
                }
                if (changed) {
                    lockInfo.known = started.flatMap((r) => r.known)
                    writeLock()
                }
            }
            sampleNow()
            timerSample = setInterval(sampleNow, treeSampleMs)
            await Promise.all(started.map((r) => r.promise))
        }
    }
    catch (err) {
        fatal = err
    }
    clearInterval(timerProgress)
    clearInterval(timerSample)

    //收尾: 樹狀終止 → 回驗埠 → 標準圖未動 → 逐實例拆除
    for (let r of started) {
        if (!r.done) {
            r.stop()
        }
    }
    await Promise.all(started.map((r) => r.promise))
    let busyPorts = []
    if (started.length > 0) {
        let knownNow = () => started.flatMap((r) => r.known)
        reap({ known: knownNow(), marker: `${MARK_ENV}=${runId}`, since: t0, phase: 'teardown' })
        let portsOfRun = started.flatMap((r) => r.rec.ports)
        let busy = await waitPortsFree(portsOfRun, 15000)
        if (busy.length > 0) {
            //仍占埠之監聽者: 開跑後才起之行程不一定是本次的(可能是他人接手該埠), 同樣須有身分證據才下殺; 無證據者由下方大聲報錯
            let extra = busy.flatMap((b) => (listenerPids(b.port, { useBoth: true, timeout: 5000, exec: ctx.exec }) || []).map((pid) => ({ pid: Number(pid), port: b.port })))
            reap({ known: knownNow(), marker: `${MARK_ENV}=${runId}`, since: t0, extra, phase: 'teardown-port' })
            busy = await waitPortsFree(busy.map((b) => b.port), 15000)
        }
        busyPorts = busy.map((b) => b.port)
        for (let b of busy) {
            teardownErrors.push(`埠 ${b.port} 未釋放: ${b.code}@${b.host}; ${describeOwners(b.port, ctx)}`)
        }
    }
    let knownCount = { current: started.flatMap((r) => r.known).length, total: knownEver.size }
    if (started.length > 0 || kills.length > 0) {
        log(`[shard] 行程清理: known 現存 ${knownCount.current} 筆(累計 ${knownCount.total}), 下殺 ${kills.length} 筆, 未下殺 ${killSkipped.length} 筆`)
    }
    let picsChanged = false
    if (picsBefore) {
        let d = snapshotLinkTarget(path.join(root, 'test', 'pics'))
        let diffKeys = Object.keys({ ...picsBefore.entries, ...d.entries }).filter((k) => picsBefore.entries[k] !== d.entries[k])
        if (diffKeys.length > 0) {
            picsChanged = true
            teardownErrors.push(`test/pics 於本次執行中被改動(工具只讀比對): ${diffKeys.slice(0, 10).join(', ')}${diffKeys.length > 10 ? ` 等 ${diffKeys.length} 項` : ''}`)
        }
    }
    for (let i of created) {
        let inst = instOf(i)
        let rec = started.find((r) => r.rec.index === i)
        if (rec && rec.rec.ports.some((p) => busyPorts.includes(p))) {
            teardownErrors.push(`實例 ${inst} 未刪除: 其埠未釋放(可能仍有行程使用該實例)`)
            continue
        }
        if (picsChanged) {
            teardownErrors.push(`實例 ${inst} 未刪除: test/pics 不完好, 留待查明(實例內含 .env 等憑證副本, 查明後請重跑本工具清除)`)
            continue
        }
        try {
            removeInst(inst, picsBefore)
        }
        catch (err) {
            teardownErrors.push(err.message)
        }
    }

    handlers.forEach(([sig, h]) => {
        if (typeof proc.removeListener === 'function') {
            proc.removeListener(sig, h)
        }
    })
    releaseLock(lock, workDir)
    for (let e of teardownErrors) {
        log(`[shard] ✘ 收尾失敗: ${e}`)
    }
    if (teardownErrors.length > 0 && fs.existsSync(workDir)) {
        log(`[shard] ✘ 工作目錄仍存在(可能含 .env 等憑證副本): ${workDir}; 查明後重跑本工具即依拆除程序清除, 切勿直接遞迴刪除或 robocopy 鏡像清除`)
    }

    if (fatal) {
        if (teardownErrors.length > 0) {
            fatal.message = `${fatal.message}\n另有收尾失敗: ${teardownErrors.join(' | ')}`
        }
        fatal.teardownErrors = teardownErrors
        fatal.kills = kills
        //有下殺即須可事後查: 例外路徑沒有彙整, 另寫 summary.json; 沒有下殺則不建立任何東西
        if (kills.length > 0) {
            try {
                fs.mkdirSync(outDir, { recursive: true })
                fs.writeFileSync(path.join(outDir, 'summary.json'), JSON.stringify({ planned: false, fatal: fatal.message, teardownErrors, kills, killSkipped, knownCount, outDir }, null, 2))
            }
            catch (err) {
                log(`[shard] 警告: 寫 summary.json 失敗(${err.message})`)
            }
        }
        throw fatal
    }

    //彙整
    let totals = { passing: 0, failing: 0, pending: 0, failures: [], incomplete: [] }
    for (let r of started) {
        let rec = r.rec
        let s = null
        try {
            s = JSON.parse(fs.readFileSync(rec.resultPath, 'utf8'))
        }
        catch (err) {}
        if (s) {
            rec.passing = s.passing
            rec.failing = s.failing
            rec.pending = s.pending
            rec.total = s.total
            rec.failures = s.failures.map((f) => ({ ...f, group: rec.index }))
            rec.fileMs = s.fileMs || {}
        }
        let ran = rec.passing + rec.failing + rec.pending
        if (!rec.note) {
            if (r.timedOut) {
                rec.note = '逾時'
            }
            else if (!s) {
                rec.note = '未產生統計'
            }
            else if (!s.done && !(s.total > 0 && ran >= s.total)) {
                rec.note = `未跑完(${ran}/${s.total})`
            }
            else if (ran === 0) {
                rec.note = '空跑(0 案例)'
            }
            else if (rec.exit !== 0 && rec.failing === 0) {
                rec.note = `mocha 離開碼 ${rec.exit === null ? `訊號 ${rec.signal}` : rec.exit} 與統計不符`
            }
        }
        rec.complete = !rec.note
        totals.passing += rec.passing
        totals.failing += rec.failing
        totals.pending += rec.pending
        totals.failures.push(...rec.failures)
        if (!rec.complete) {
            totals.incomplete.push(rec.index)
        }
    }
    //未開跑之組(收到訊號時)亦屬未跑完
    for (let g of plan.groups) {
        if (!started.some((r) => r.rec.index === g.index)) {
            totals.incomplete.push(g.index)
        }
    }

    let wallMs = Date.now() - t0
    let durations = {}
    for (let r of started) {
        for (let [f, ms] of Object.entries(r.rec.fileMs)) {
            durations[f] = min2(ms)
        }
    }

    log('[shard] ── 各組結果 ──')
    for (let r of started) {
        let c = r.rec
        log(`[shard] 第 ${c.index} 組 位移 ${c.offset} ${min2(c.ms)} 分 通過 ${c.passing} 失敗 ${c.failing} pending ${c.pending}${c.note ? ` ✘${c.note}` : ''}: ${c.files.join(', ')}`)
    }
    log('')
    log(`  ${totals.passing} passing (${fmtWall(wallMs)})`)
    log(`  ${totals.failing} failing`)
    log(`  ${totals.pending} pending`)
    log('')
    totals.failures.forEach((f, i) => {
        log(`  ${i + 1}) [第${f.group}組${f.file ? ` ${f.file}` : ''}] ${f.title}`)
    })
    if (totals.incomplete.length > 0) {
        log(`[shard] ✘ 未跑完之組: ${[...new Set(totals.incomplete)].join(', ')}`)
    }
    if (Object.keys(durations).length > 0) {
        log('[shard] durations 實測值(分; 可貼回 e2eShard.json):')
        log(JSON.stringify({ durations }, null, 4))
    }
    if (opt.updateDurations === true && Object.keys(durations).length > 0) {
        let { cfg: cur } = readShardConfig(fpConfig)
        cur.durations = { ...(isObj(cur.durations) ? cur.durations : {}), ...durations }
        fs.mkdirSync(path.dirname(fpConfig), { recursive: true })
        fs.writeFileSync(fpConfig, `${JSON.stringify(cur, null, 4)}\n`)
        log(`[shard] 已寫回 durations: ${fpConfig}`)
    }

    let exitCode = 0
    if (signal) {
        exitCode = SIGNAL_CODES[signal] || 1
    }
    else if (totals.failing > 0 || totals.incomplete.length > 0 || teardownErrors.length > 0) {
        exitCode = 1
    }
    let groupsOut = started.map((r) => r.rec)
    let result = {
        planned: false,
        exitCode,
        signal,
        plan,
        groups: groupsOut,
        totals,
        wallMs,
        durations,
        outDir,
        teardownErrors,
        kills,
        killSkipped,
        knownCount,
    }
    try {
        fs.mkdirSync(outDir, { recursive: true })
        fs.writeFileSync(path.join(outDir, 'summary.json'), JSON.stringify(result, null, 2))
    }
    catch (err) {
        log(`[shard] 警告: 寫 summary.json 失敗(${err.message})`)
    }
    log(`[shard] log 與統計: ${outDir}`)
    log(`[shard] ${exitCode === 0 ? '✔ 全部通過, 實例已移除' : `✘ 離開碼 ${exitCode}`}`)
    return result
}
