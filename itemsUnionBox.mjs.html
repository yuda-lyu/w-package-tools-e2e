/**
 * 紅框「可見項目之聯集」之量測型目標：清單、樹、選單、訊息時間軸、工具列上之項目、一行訊息文字等，框實際有內容之項目而非整欄/整區
 *
 * 交給captureStableWithBox作為target(或其陣列之一員)。技能 §7.2「清單/樹重新列出、展開縮合、進入頁面 → 當時可見之項目列的聯集」、
 * §7.3-2「不框整欄/整區的空白」。量測於捲入與等待之後、截圖之前進行。
 * 項目：選擇器字串、其陣列(以逗號併為一個選擇器)，或Playwright Locator(例：getByText 取得之訊息文字)；選擇器與Locator皆經同一個頁內量測函式。
 * 可見項目＝命中項目、寬高大於0、非display:none/visibility:hidden/opacity:0，且與範圍容器(within；未給則為視窗)有交集者；
 * 聯集再夾在範圍容器內(容器可捲動時，捲出可視範圍之項目不計入)。一個都量不到時回傳null(captureStableWithBox因此拋錯)。
 * fit：依「元素自身有無可見邊界」決定量法——有(底色異於其背後底色、至少三邊之邊框、陰影、背景圖，或圖片/表單等替換元素；晶片、按鈕、
 * 灰頁上之白卡、底色列)量元素本身；無(透明容器、白頁上之白底區塊、整行文字、開關與其標籤、勾選列、無內距之欄位與表單欄)
 * 改量其可見內容(有可見邊界之子元素框與文字行框之聯集；透明子元素之內距不計)，且一律外擴 inkPad。
 * 用於「一行字卻框整列寬」(技能 §7.2 提示訊息/狀態文字→訊息本體)、「一列晶片卻框整列寬」與「無邊界之元件緊貼文字」。
 * inkPad(2026-09-28)：紅框幾何為目標 ±6 且 5px 描邊內縮，無可見邊界之目標緊貼墨跡時框內僅剩約 1px、紅框壓字
 * (SSO login E2E-006、stainfor E2E-002、tokens 權限勾選列與編輯模式開關、連線狀態文字殷鑑)；外擴 4px 後框內留白約 5px，與自帶內距之按鈕相當。
 * 子元素(勾選框、開關軌道、圖示、輸入框)與文字同外擴(同日改：原只外擴文字，子元素框貼框線 0–1px、四邊留白不一致)。
 * fit 下框線不壓框外內容(2026-09-28，業主：「亂框亂壓導致遮蔽有效資訊或使用者誤解就是問題」)：
 * ①相鄰項目(自元素往上至多 4 層之緊鄰兄弟，上下左右四向各取最近一層)：框線置於兩者**可見範圍**(有可見邊界者為元素框，否則為子元素與文字之聯集)
 *   間隙正中，不壓到鄰項之字(例：權限清單緊鄰列、模式段與名稱段)，也不因鄰項元素框之透明內距而貼回目標自身之字(例：開關右側之圖示鈕)；
 *   無可見內容之兄弟(空白間隔)不算鄰項；
 * ②浮層(自身或祖先為 position:fixed，例：下拉清單、浮出面板；全畫面者除外)：浮層外之頁面內容不是 DOM 兄弟，改以命中測試找目標四周、
 *   浮層範圍外最近之可見內容(文字、晶片／按鈕／觸發區級之有邊界元件、圖示與表單元件；大容器之底色與整張圖表不算；遮罩後之背景不算)，
 *   框線置於與其間隙正中；浮層本身為目標時以其內容之可見範圍計，框線可退入浮層內距；四周無內容者照常外擴
 *   (同日首版一律畫在浮層內側，內距小之清單框線反而貼字：PERM 時間清單下緣 2px、關聯模式清單左緣 1px)。
 * ③圖形內容(2026-10-20，Gitea issue #6：浮層疊在大型 svg/canvas 圖表上，框線蓋到其下之資料波形)：②之命中測試遇祖先寬 > 480 或高 > 120 即中止，
 *   故大型 svg 內之圖形與 canvas 像素原判為無內容。graphics(預設開)另以不依賴命中測試之方式找浮層四周之圖形墨跡——
 *   svg 圖形(path/line/polyline/polygon/rect/circle/ellipse/text)以 getScreenCTM 反矩陣 + isPointInStroke/isPointInFill 幾何判定
 *   (pointer-events:none 之 Plotly、opacity:0 或全透明追蹤線之 Highcharts 皆正確)，可見性含累積 opacity、stroke/fill 透明度、漸層色標、祖先 overflow 裁切、
 *   clip-path、被有底色之層蓋住；撐滿圖表之 rect 與填色同背後底色者(圖表背景)不算內容；2d canvas 以 drawImage 分塊讀像素(不對來源呼叫 getContext)，
 *   與其所在 16x16 小格之背景色有對比者為墨跡(均勻大色塊視為背景)。
 *   只增不減：②回非 null 之點結果不變，圖形規則另行掃描、逐邊取較近者；只在框線帶(目標矩形邊外 1 至 6px，另含緊鄰矩形邊之一列/行作守距)內有墨跡時才介入，
 *   沿矩形邊全長逐 px 取樣，四向同一算法；矩形內(含 inkPad 區)之內容框線蓋不到，不介入(框線四角 6x6 不掃，角落處孤立之內容為已知限制)；
 *   候選範圍為浮層外層矩形與框矩形之聯集外擴 64px。框線外緣退到內容前 1px(剛好不蓋；不似 HTML 鄰項取間隙正中——圖形遍布整個繪圖區，浮層又常比其文字寬，
 *   取正中會使框縮到只剩文字)，框內緣不越過文字，放不下 5px 描邊時退回間隙正中；圖形夾制使框夾到沒有尺寸時只撤銷圖形之夾制(HTML 規則之結果保留)。
 *   寬與高皆不大於 480x120 之小 svg/canvas(圖示、迷你圖)②已以整個元素框為內容，不重複判定。
 *   不自動判定者(已知限制)：img、video、WebGL(未保留繪圖緩衝區者讀得空白)與跨域污染之 canvas、含旋轉/翻轉變換之 canvas、object-position 非置中之 object-fit canvas、
 *   svg 之 use/image/marker、shadow DOM、iframe 內之圖表，以 graphicAreas 宣告整區視為內容。graphics:false 逐位元重現 1.0.9 之行為。
 *   會動使用端標準圖：fixed 浮層之框線帶內有圖形墨跡者框線改退入，使用端須重產受影響之圖並逐張審。
 *
 * @param {String|Array|Object} itemSel 輸入項目之選擇器字串、其陣列，或Playwright Locator
 * @param {Object} [opt={}] 輸入設定物件，預設{}
 * @param {String} [opt.within=''] 輸入範圍容器之選擇器字串(項目於其內查找，聯集夾在其內)，取第一個命中者，預設''代表整頁與視窗
 * @param {String|Object} [opt.scroll] 輸入先捲入視窗之選擇器字串或Locator，預設同within；within亦未給時，項目為Locator者捲該Locator，否則不捲動
 * @param {Boolean} [opt.fit=false] 輸入無可見邊界之項目是否改量其內容範圍並外擴文字墨跡布林值，預設false
 * @param {Number} [opt.inkPad=4] 輸入fit下文字墨跡之外擴像素，預設4
 * @param {Boolean} [opt.graphics=true] 輸入fit下浮層四周是否判定svg圖形與可讀2d canvas像素為框外內容布林值，預設true；false逐位元重現1.0.9
 * @param {String|Array} [opt.graphicAreas=null] 輸入整區視為框外內容之選擇器字串或其陣列(img、video、WebGL或跨域污染之canvas、地圖、熱圖等無從自動判定者)，預設null；空字串或非字串項拋錯
 * @returns {Object} 回傳量測型目標物件{label,scroll,measure}
 * @example
 *
 * import itemsUnionBox from 'w-package-tools-e2e/src/itemsUnionBox.mjs'
 *
 * let buf = await captureStableWithBox(page, itemsUnionBox('.tree-item', { within: '[data-fmid="channel-tree"]' }))
 * let buf2 = await captureStableWithBox(page, ['img.spinner', itemsUnionBox(page.getByText('Connecting...').first(), { fit: true })])
 * //下拉清單浮在Highcharts/ECharts圖表上：框線自動退到不蓋資料線(graphics預設開)；浮在地圖(WebGL)上則宣告地圖區為內容
 * let buf3 = await captureStableWithBox(page, itemsUnionBox('.WPopperFix:visible', { fit: true, graphicAreas: '#map canvas' }))
 *
 */
function itemsUnionBox(itemSel, opt = {}) {
    let { within = '', scroll, fit = false, inkPad = INK_PAD, graphics = true, graphicAreas = null } = opt
    let areaSels = graphicAreas === null || graphicAreas === undefined || graphicAreas === false ? [] : (Array.isArray(graphicAreas) ? graphicAreas : [graphicAreas])
    if (!areaSels.every((s) => typeof s === 'string' && s.trim() !== '')) {
        throw new Error('itemsUnionBox: graphicAreas 須為非空選擇器字串或其陣列')
    }
    let isLoc = !!itemSel && typeof itemSel === 'object' && !Array.isArray(itemSel) && typeof itemSel.evaluateAll === 'function'
    let sel = isLoc ? '' : (Array.isArray(itemSel) ? itemSel.join(', ') : itemSel)
    if (!isLoc && (typeof sel !== 'string' || sel.trim() === '')) {
        throw new Error('itemsUnionBox: itemSel 須為非空之選擇器字串、其陣列或 Locator')
    }
    if (typeof inkPad !== 'number' || !(inkPad >= 0)) {
        throw new Error('itemsUnionBox: inkPad 須為非負數')
    }
    if (scroll === undefined) {
        scroll = within || (isLoc ? itemSel : null)
    }
    return {
        label: `itemsUnionBox(${isLoc ? 'Locator' : sel}${within ? ` within ${within}` : ''}${fit ? ' fit' : ''})`,
        scroll,
        probe: isLoc ? itemSel : null, //項目為單一 Locator 時供 captureStableWithBox 做被蓋住檢查(同直接傳 Locator)
        measure: async (page) => {
            let loc = isLoc ? itemSel : (within ? page.locator(within).first().locator(sel) : page.locator(sel))
            //頁內量測(選擇器與 Locator 共用此一函式)
            let measureEls = (els, { within, fit, inkPad, edge, strokeHalf, debug, graphics, areaSels }) => {
                let dbg = []
                let uni = (a, b) => (!a ? b : !b ? a : { left: Math.min(a.left, b.left), top: Math.min(a.top, b.top), right: Math.max(a.right, b.right), bottom: Math.max(a.bottom, b.bottom) })
                //元素自身有可見邊界(底色異於其背後底色、邊框、陰影、背景圖，或圖片/表單等替換元素)：其邊界即視覺邊界
                let rgbaOf = (c) => {
                    let m = String(c || '').match(/rgba?\(([^)]+)\)/)
                    if (!m) {
                        return (c && c !== 'transparent') ? [0, 0, 0, 1] : [0, 0, 0, 0]
                    }
                    let p = m[1].split(/[\s,/]+/).filter((s) => s !== '').map(Number)
                    return [p[0], p[1], p[2], p.length >= 4 ? p[3] : 1]
                }
                let backdropOf = (e) => {
                    for (let p = e.parentElement; p; p = p.parentElement) {
                        let c = rgbaOf(window.getComputedStyle(p).backgroundColor)
                        if (c[3] > 0) {
                            return c
                        }
                    }
                    return [255, 255, 255, 1]
                }
                let hasBoundary = (e) => {
                    if (['IMG', 'SVG', 'CANVAS', 'VIDEO', 'INPUT', 'BUTTON', 'SELECT', 'TEXTAREA'].includes(e.tagName.toUpperCase())) {
                        return true
                    }
                    let cs = window.getComputedStyle(e)
                    if ((cs.backgroundImage && cs.backgroundImage !== 'none') || (cs.boxShadow && cs.boxShadow !== 'none')) {
                        return true
                    }
                    let bg = rgbaOf(cs.backgroundColor)
                    if (bg[3] > 0) {
                        let bd = backdropOf(e)
                        if ([0, 1, 2].some((i) => Math.abs(bg[i] - bd[i]) > 8)) {
                            return true //底色與背後不同(白底卡片放在白頁上則不算)
                        }
                    }
                    //邊框須至少三邊可見才圍成框；單邊或上下兩條為分隔線(例：工具列 border-top)，不算
                    return ['Top', 'Right', 'Bottom', 'Left'].filter((s) => parseFloat(cs[`border${s}Width`]) > 0 && cs[`border${s}Style`] !== 'none' && rgbaOf(cs[`border${s}Color`])[3] > 0).length >= 3
                }
                //可見範圍：有可見邊界者為元素框；否則為其可見子元素之可見範圍與文字行框之聯集(遞迴，overflow 非 visible 者夾在元素框內)；
                //只以偽元素呈現者(icon font 之 ::before/::after)為元素框；無任何可見內容(空白間隔、透明占位)回 null。
                //每次計算走訪元素數設上限，超過者以元素框計(保守，同未細分)。contentOnly：不看元素自身邊界、只取其內容(浮層本身為目標時用)
                //structRef(浮層面板)：其內撐滿面板寬或高(≥ 90%)之有邊界子元素(標頭列、清單底色)為結構性底色，不以其框計、改看其內之文字與元件
                //(否則全寬標頭使內容範圍貼到面板邊，框線無處可退而壓到緊貼浮層之背景按鈕：tokens 權限浮層右緣殷鑑)
                let budget = 0
                let structRef = null
                let visExtent = (e, contentOnly = false, ref = null) => {
                    budget = 3000
                    structRef = ref
                    let v = visRec(e, contentOnly)
                    structRef = null
                    return v
                }
                let visRec = (e, contentOnly = false) => {
                    let r = e.getBoundingClientRect()
                    let structural = !!structRef && (r.width >= structRef.width * 0.9 || r.height >= structRef.height * 0.9)
                    if ((!contentOnly && !structural && hasBoundary(e)) || --budget < 0) {
                        return (r.width > 0 && r.height > 0) ? r : null
                    }
                    let acc = null
                    for (let n of e.childNodes) {
                        if (n.nodeType === 3) {
                            if (!n.nodeValue || n.nodeValue.trim() === '') {
                                continue
                            }
                            let rg = document.createRange()
                            rg.selectNodeContents(n)
                            for (let q of rg.getClientRects()) {
                                if (q.width > 0 && q.height > 0) {
                                    acc = uni(acc, q)
                                }
                            }
                        }
                        else if (n.nodeType === 1) {
                            let cs = window.getComputedStyle(n)
                            if (cs.display === 'none' || cs.visibility === 'hidden' || Number(cs.opacity) === 0) {
                                continue
                            }
                            let q = visRec(n)
                            if (q) {
                                acc = uni(acc, q)
                            }
                        }
                    }
                    if (!acc) {
                        let pseudo = ['::before', '::after'].some((p) => {
                            let c = window.getComputedStyle(e, p).content
                            return !!c && !['none', 'normal', '""', '\'\''].includes(c)
                        })
                        return (pseudo && r.width > 0 && r.height > 0) ? r : null
                    }
                    let cs = window.getComputedStyle(e)
                    if (cs.overflowX !== 'visible' || cs.overflowY !== 'visible') {
                        acc = { left: Math.max(acc.left, r.left), top: Math.max(acc.top, r.top), right: Math.min(acc.right, r.right), bottom: Math.min(acc.bottom, r.bottom) }
                        if (acc.right <= acc.left || acc.bottom <= acc.top) {
                            return null
                        }
                    }
                    return acc
                }
                let sameRect = (a, b, tol) => Math.abs(a.left - b.left) <= tol && Math.abs(a.top - b.top) <= tol && Math.abs(a.right - b.right) <= tol && Math.abs(a.bottom - b.bottom) <= tol
                //自身或後代(至多 6 層)中，矩形與 rc 相同(容差 tol)之有邊界元素：透明外層包著之面板、按鈕等
                let boundedFilling = (root, rc, tol) => {
                    let found = null
                    let walk = (el, depth) => {
                        if (found || depth > 6) {
                            return
                        }
                        if (hasBoundary(el) && sameRect(el.getBoundingClientRect(), rc, tol)) {
                            found = el
                            return
                        }
                        for (let c of el.children) {
                            walk(c, depth + 1)
                        }
                    }
                    walk(root, 0)
                    return found
                }
                //浮層外可見內容之命中測試：點在浮層範圍內者(背後內容被浮層蓋住)、視窗外者不算；全畫面透明層(點擊攔截)略過往下找，
                //全畫面有底色之遮罩(背後已調暗)不算內容。命中元素往上找：文字行框含此點者、有可見邊界之小元件(寬 ≤ 480 且高 ≤ 120：
                //晶片、按鈕、觸發區、輸入框、圖示與表單等替換元素)為內容並回傳其矩形；遇大容器即止(大片底色、整張圖表不算)
                let textRectAt = (a, x, y) => {
                    for (let n of a.childNodes) {
                        if (n.nodeType !== 3 || !n.nodeValue || n.nodeValue.trim() === '') {
                            continue
                        }
                        let rg = document.createRange()
                        rg.selectNodeContents(n)
                        for (let q of rg.getClientRects()) {
                            if (x >= q.left && x <= q.right && y >= q.top && y <= q.bottom) {
                                return q
                            }
                        }
                    }
                    return null
                }
                //圖形內容(svg 圖形、canvas 像素、使用端宣告之繪圖區)：與舊規則(elementsFromPoint 取命中元素)分開、另以幾何與像素判定，兩者結果取較近者(只增不減)。
                //svg 圖形：getScreenCTM 反矩陣 + isPointInStroke/isPointInFill，不依賴命中測試——pointer-events:none 之圖形(Plotly 之 .main-svg)命中不到、
                //opacity:0 之圖形與全透明追蹤線(Highcharts 之 tracker)命中得到卻看不見；canvas：以 drawImage 取其像素至離屏小畫布(不對來源呼叫 getContext，
                //免替從未繪製之 canvas 建立 2d context 而使其後之 webgl 失效)；img/video/跨域污染之 canvas(WebGL 未保留繪圖緩衝區者讀得空白)無從自動判定，
                //由 graphicAreas 宣告整區視為內容
                let gnear = {} //debug 用：各邊命中之最近圖形(描述@視窗座標)
                let gLast = ''
                let gNote = (d) => {
                    gLast = d
                }
                let gDesc = (e) => `${e.tagName.toLowerCase()}${e.id ? `#${e.id}` : ''}${(typeof e.className === 'string' ? e.className : (e.className && e.className.baseVal) || '').trim().split(/\s+/).filter(Boolean).slice(0, 2).map((c) => `.${c}`).join('')}`
                let G_MIN_ALPHA = 0.1
                let rectInter = (a, b) => (!a ? b : !b ? a : { left: Math.max(a.left, b.left), top: Math.max(a.top, b.top), right: Math.min(a.right, b.right), bottom: Math.min(a.bottom, b.bottom) })
                //顏色解析(圖形規則專用)：計算樣式之色多為 rgb()/rgba()，但 color(srgb|display-p3 …)、lab()、oklch() 等現代色彩空間不在此列——
                //舊 rgbaOf 把不認得者一律當不透明黑，使全透明之 P3 色線被當內容。alpha 取「/ a」之 a(可為百分比)；rgb 僅 rgb()、color(srgb …) 可得，其餘為 null(不做底色比對)
                let gColor = (c) => {
                    let s = String(c || '').trim()
                    if (s === '' || s === 'none' || s === 'transparent') {
                        return { rgb: null, a: 0 }
                    }
                    let num = (t) => (String(t).endsWith('%') ? parseFloat(t) / 100 : Number(t))
                    let m = s.match(/^rgba?\(([^)]+)\)$/)
                    if (m) {
                        let p = m[1].split(/[\s,/]+/).filter((t) => t !== '')
                        let a = p.length >= 4 ? num(p[3]) : 1
                        return { rgb: [Number(p[0]), Number(p[1]), Number(p[2])], a: Number.isNaN(a) ? 1 : a }
                    }
                    let ma = s.match(/\/\s*([\d.]+%?)\s*\)\s*$/)
                    let a = ma ? num(ma[1]) : 1
                    let ms = s.match(/^color\(\s*srgb\s+([-\d.e]+)\s+([-\d.e]+)\s+([-\d.e]+)/)
                    return { rgb: ms ? [Number(ms[1]) * 255, Number(ms[2]) * 255, Number(ms[3]) * 255] : null, a: Number.isNaN(a) ? 1 : a }
                }
                //自元素至文件根之累積 opacity、是否經 clip-path、是否含旋轉/翻轉/斜切之 CSS 變換、overflow 非 visible 之祖先矩形交集(裁切範圍)
                //(以節點為鍵記憶，千個圖形共用祖先之計算)
                let gUp = new Map()
                let gUpOf = (n) => {
                    if (!n || n.nodeType !== 1) {
                        return { op: 1, clip: false, warp: false, box: null }
                    }
                    let m = gUp.get(n)
                    if (!m) {
                        let cs = window.getComputedStyle(n)
                        let up = gUpOf(n.parentElement)
                        let warp = up.warp
                        if (!warp && cs.transform && cs.transform !== 'none') {
                            try {
                                let mx = new DOMMatrix(cs.transform)
                                warp = Math.abs(mx.b) > 1e-6 || Math.abs(mx.c) > 1e-6 || mx.a < 0 || mx.d < 0
                            }
                            catch (err) {
                                warp = true
                            }
                        }
                        let box = up.box
                        if ((cs.overflowX !== 'visible' || cs.overflowY !== 'visible') && !n.ownerSVGElement) { //巢狀 svg 之視埠矩形無從由 getBoundingClientRect 取得(回內容外接框)，不裁
                            let r = n.getBoundingClientRect()
                            box = rectInter(box, { left: r.left, top: r.top, right: r.right, bottom: r.bottom })
                        }
                        m = { op: Number(cs.opacity) * up.op, clip: up.clip || (!!cs.clipPath && cs.clipPath !== 'none'), warp, box }
                        gUp.set(n, m)
                    }
                    return m
                }
                let gVisible = (e) => {
                    let cs = window.getComputedStyle(e)
                    return cs.display !== 'none' && cs.visibility !== 'hidden' && gUpOf(e).op >= G_MIN_ALPHA
                }
                //漸層/圖樣填色(url(#id))：參考不存在視同 none；漸層之色標全為透明視同 none；其餘視為可見
                let gServerAlpha = (v) => {
                    let mt = v.match(/url\(\s*["']?#([^"')\s]+)/)
                    let ref = mt ? document.getElementById(mt[1]) : null
                    if (!ref) {
                        return 0
                    }
                    let tn = ref.tagName.toLowerCase()
                    if (tn !== 'lineargradient' && tn !== 'radialgradient') {
                        return 1
                    }
                    let stops = ref.querySelectorAll('stop')
                    if (stops.length === 0) {
                        let hr = ref.getAttribute('href') || ref.getAttribute('xlink:href')
                        let tg = hr && hr.startsWith('#') ? document.getElementById(hr.slice(1)) : null
                        stops = tg ? tg.querySelectorAll('stop') : []
                    }
                    let best = 0
                    for (let s of stops) {
                        let scs = window.getComputedStyle(s)
                        best = Math.max(best, gColor(scs.stopColor).a * Number(scs.stopOpacity))
                    }
                    return best
                }
                let gPaint = (v, opa, acc) => {
                    if (!v || v === 'none') {
                        return 0
                    }
                    return (v.startsWith('url(') ? gServerAlpha(v) : gColor(v).a) * opa * acc
                }
                let gCtxs = new Map()
                //掃描範圍(目標與浮層外層矩形之聯集外擴 64px，見 nearestOutside)內之候選：與之相交之 svg 根(及 overflow 可見者：內容可畫在根框外)、canvas、使用端宣告之繪圖區；
                //每次量測只掃描全頁一遍，之後逐取樣點只查此小清單(頁面有數萬個 svg 圖示時不致逐點全掃)
                let gCtxOf = (region) => {
                    let key = `${region.left},${region.top},${region.right},${region.bottom}`
                    let c = gCtxs.get(key)
                    if (!c) {
                        let near = (r) => r.width > 0 && r.height > 0 && r.right >= region.left && r.left <= region.right && r.bottom >= region.top && r.top <= region.bottom
                        c = { roots: [], canvases: [], areas: [], shapes: new Map(), canvas: new Map() }
                        //圖形規則補的是舊規則「祖先寬 > 480 或高 > 120 即中止」之盲區：寬 ≤ 480 且高 ≤ 120 之小 svg/canvas(圖示、迷你圖)舊規則已以整個元素框為內容
                        //(hasBoundary)，不在此重複(頁面有數萬個 svg 圖示時亦免逐個列舉)。svg 根：與範圍相交之大者收；宣告 overflow:visible(內容可畫在根框外)者收——
                        //小根只認屬性與行內樣式(不對每個圖示取計算樣式)，大根另認樣式表規則(計算樣式)
                        let big = (r) => r.width > 480 || r.height > 120
                        for (let s of document.querySelectorAll('svg')) {
                            if (s.ownerSVGElement) {
                                continue
                            }
                            let sr = s.getBoundingClientRect()
                            let isBig = big(sr)
                            let visibleOverflow = s.getAttribute('overflow') === 'visible' || s.style.overflow === 'visible' || s.style.overflowX === 'visible' || s.style.overflowY === 'visible'
                            if (!visibleOverflow && isBig && !near(sr)) {
                                let scs = window.getComputedStyle(s)
                                visibleOverflow = scs.overflowX === 'visible' || scs.overflowY === 'visible'
                            }
                            if (((near(sr) && isBig) || visibleOverflow) && window.getComputedStyle(s).display !== 'none') {
                                c.roots.push(s)
                            }
                        }
                        for (let cv of document.querySelectorAll('canvas')) {
                            let cr = cv.getBoundingClientRect()
                            if (near(cr) && big(cr)) {
                                c.canvases.push(cv)
                            }
                        }
                        for (let sel of areaSels) {
                            for (let a of document.querySelectorAll(sel)) {
                                if (near(a.getBoundingClientRect())) {
                                    c.areas.push(a)
                                }
                            }
                        }
                        gCtxs.set(key, c)
                    }
                    return c
                }
                //一個 svg 根內與掃描範圍相交之可見圖形。bbox 不含線寬：先以 200px 寬之預篩(線寬 > 400px 者不認)，取得線寬後再以精確之 pad 複核
                let gShapesOf = (ctx, root, region) => {
                    let cached = ctx.shapes.get(root)
                    if (cached) {
                        return cached
                    }
                    let rr = root.getBoundingClientRect()
                    let rootArea = rr.width * rr.height
                    let all = [...root.querySelectorAll('path,line,polyline,polygon,rect,circle,ellipse,text')].map((el) => ({ el, q0: el.getBoundingClientRect() }))
                    //圖表自身之底色：最上層撐滿(面積 ≥ 根面積 50%)且不透明之 rect 之色(深色主題之黑底：白色形狀與黑底不同，不可與頁面底色比)；無則取根之祖先底色
                    let svgBg = backdropOf(root).slice(0, 3)
                    for (let { el, q0 } of all) {
                        if (el.tagName.toLowerCase() === 'rect' && q0.width * q0.height >= rootArea * 0.5 && !el.closest('defs,clipPath,mask,pattern,marker,symbol')) {
                            let bcs = window.getComputedStyle(el)
                            let bc = gColor(bcs.fill)
                            if (bc.rgb && bc.a * Number(bcs.fillOpacity) >= 0.9 && bcs.display !== 'none' && bcs.visibility !== 'hidden') {
                                svgBg = bc.rgb
                            }
                        }
                    }
                    let list = []
                    for (let { el, q0 } of all) {
                        if (q0.right + 200 < region.left || q0.left - 200 > region.right || q0.bottom + 200 < region.top || q0.top - 200 > region.bottom) {
                            continue
                        }
                        if (el.closest('defs,clipPath,mask,pattern,marker,symbol,linearGradient,radialGradient,filter')) {
                            continue
                        }
                        let cs = window.getComputedStyle(el)
                        if (cs.display === 'none' || cs.visibility === 'hidden') {
                            continue
                        }
                        let up = gUpOf(el)
                        let acc = up.op
                        let m = el.getScreenCTM()
                        if (!m || acc < G_MIN_ALPHA) {
                            continue
                        }
                        let inv = null
                        try {
                            inv = m.inverse()
                        }
                        catch (err) {
                            continue
                        }
                        let tn = el.tagName.toLowerCase()
                        let isText = tn === 'text'
                        let sw = parseFloat(cs.strokeWidth)
                        let strokeOK = !isText && sw > 0 && gPaint(cs.stroke, Number(cs.strokeOpacity), acc) >= G_MIN_ALPHA
                        let fillA = gPaint(cs.fill, Number(cs.fillOpacity), acc)
                        let fillOK = !isText && fillA >= G_MIN_ALPHA
                        if (fillOK) {
                            //底色：撐滿圖表之矩形(背景、繪圖區底色，任何顏色)、或填色與圖表底色相同(白底圖表上之白色形狀：背景 path、白色遮片)皆不算內容；其邊框仍算
                            //(同舊規則之「大型容器之底色不算內容」「底色與背後相同則無可見邊界」)
                            if (tn === 'rect' && q0.width * q0.height >= rootArea * 0.5) {
                                fillOK = false
                            }
                            else if (!cs.fill.startsWith('url(')) {
                                let fc = gColor(cs.fill)
                                if (fc.rgb) {
                                    let k = Math.min(1, fillA)
                                    if ([0, 1, 2].every((i) => Math.abs((fc.rgb[i] * k + svgBg[i] * (1 - k)) - svgBg[i]) <= 8)) {
                                        fillOK = false
                                    }
                                }
                            }
                        }
                        let sc = Math.max(Math.hypot(m.a, m.b), Math.hypot(m.c, m.d))
                        let pad = (sw > 0 ? sw * sc * 1.5 : 0) + 1
                        let rects = null
                        let q = { left: q0.left - pad, top: q0.top - pad, right: q0.right + pad, bottom: q0.bottom + pad }
                        if (isText) {
                            //文字取各文字節點之字行框(同舊規則之 textRectAt)，並依該節點所屬元素(text 或 tspan)自身之樣式判可見(fill 或 stroke 有可見色、非透明、非隱藏)；
                            //舊規則只認命中測試得到者，此處不依賴命中測試：被透明層蓋住、pointer-events:none 之文字亦算
                            rects = []
                            let tw = document.createTreeWalker(el, NodeFilter.SHOW_TEXT)
                            for (let tnode = tw.nextNode(); tnode; tnode = tw.nextNode()) {
                                let pe = tnode.parentElement
                                if (!tnode.nodeValue || tnode.nodeValue.trim() === '' || !pe) {
                                    continue
                                }
                                let pcs = window.getComputedStyle(pe)
                                let pacc = gUpOf(pe).op
                                if (pcs.display === 'none' || pcs.visibility === 'hidden' || pacc < G_MIN_ALPHA) {
                                    continue
                                }
                                if (gPaint(pcs.fill, Number(pcs.fillOpacity), pacc) < G_MIN_ALPHA && !(parseFloat(pcs.strokeWidth) > 0 && gPaint(pcs.stroke, Number(pcs.strokeOpacity), pacc) >= G_MIN_ALPHA)) {
                                    continue
                                }
                                let rg = document.createRange()
                                rg.selectNodeContents(tnode)
                                rects.push(...[...rg.getClientRects()].filter((r) => r.width > 0 && r.height > 0))
                            }
                            if (rects.length === 0) {
                                continue
                            }
                            q = { left: Math.min(...rects.map((r) => r.left)) - 1, top: Math.min(...rects.map((r) => r.top)) - 1, right: Math.max(...rects.map((r) => r.right)) + 1, bottom: Math.max(...rects.map((r) => r.bottom)) + 1 }
                        }
                        else if (!strokeOK && !fillOK) {
                            continue
                        }
                        if (q.right < region.left || q.left > region.right || q.bottom < region.top || q.top > region.bottom) {
                            continue
                        }
                        //經 clip-path 者(Highcharts 之繪圖區裁切)：圖形可能被裁掉看不見，能命中測試者(pointer-events 非 none)以命中堆疊確認
                        list.push({ el, isText, rects, strokeOK, fillOK, inv, box: up.box, clip: up.clip && cs.pointerEvents !== 'none', q })
                    }
                    ctx.shapes.set(root, list)
                    return list
                }
                //canvas 像素：以 drawImage 分塊(64x64 裝置像素)取至離屏小畫布按需讀取(不整張讀，地圖級 canvas 數百萬像素；不對來源呼叫 getContext)；
                //每個 canvas 各用一張離屏畫布(跨域污染之 canvas 會污染其離屏畫布，不可與其他 canvas 共用)；背景色取各像素所在 16x16 小格之眾數(見 gCanvasInk)。
                //座標對應須扣 border 與 padding(位圖只畫在內容盒)並依 object-fit 換算；含旋轉/翻轉之 CSS 變換者、object-position 非置中者與跨域污染者無從判定
                let gCanvasOf = (ctx, c) => {
                    if (ctx.canvas.has(c)) {
                        return ctx.canvas.get(c)
                    }
                    let st = null
                    try {
                        let up = gUpOf(c)
                        let cs = window.getComputedStyle(c)
                        if (gVisible(c) && !up.warp && c.width > 0 && c.height > 0) {
                            let r = c.getBoundingClientRect()
                            let px = (k) => parseFloat(cs[k]) || 0
                            let cb = { left: r.left + px('borderLeftWidth') + px('paddingLeft'), top: r.top + px('borderTopWidth') + px('paddingTop'), width: r.width - px('borderLeftWidth') - px('paddingLeft') - px('borderRightWidth') - px('paddingRight'), height: r.height - px('borderTopWidth') - px('paddingTop') - px('borderBottomWidth') - px('paddingBottom') }
                            let fit = cs.objectFit || 'fill'
                            let map = null
                            if (cb.width > 0 && cb.height > 0) {
                                if (fit === 'fill') {
                                    map = { ox: cb.left, oy: cb.top, dw: cb.width, dh: cb.height, sx: c.width / cb.width, sy: c.height / cb.height }
                                }
                                else if (['contain', 'cover', 'none', 'scale-down'].includes(fit) && cs.objectPosition === '50% 50%') {
                                    let k = fit === 'contain' ? Math.min(cb.width / c.width, cb.height / c.height) : fit === 'cover' ? Math.max(cb.width / c.width, cb.height / c.height) : fit === 'none' ? 1 : Math.min(1, cb.width / c.width, cb.height / c.height)
                                    let dw = c.width * k
                                    let dh = c.height * k
                                    map = { ox: cb.left + (cb.width - dw) / 2, oy: cb.top + (cb.height - dh) / 2, dw, dh, sx: 1 / k, sy: 1 / k }
                                }
                            }
                            if (map) {
                                let scratch = document.createElement('canvas')
                                scratch.width = 64
                                scratch.height = 64
                                st = { c, w: c.width, h: c.height, tiles: new Map(), box: up.box, clip: up.clip && cs.pointerEvents !== 'none', map, scratch }
                            }
                        }
                    }
                    catch (err) {
                        st = null //無從自動判定
                    }
                    ctx.canvas.set(c, st)
                    return st
                }
                //讀 canvas 位圖座標(cx, cy)之像素：回傳 { d(該塊資料), o(位移), t(塊), lx, ly } 或 null(無法讀取)
                let gCanvasRead = (st, cx, cy) => {
                    let tx = Math.floor(cx / 64)
                    let ty = Math.floor(cy / 64)
                    let k = `${tx},${ty}`
                    let t = st.tiles.get(k)
                    if (t === undefined) {
                        try {
                            let tw = Math.min(64, st.w - tx * 64)
                            let th = Math.min(64, st.h - ty * 64)
                            let sx = st.scratch.getContext('2d', { willReadFrequently: true })
                            sx.clearRect(0, 0, 64, 64)
                            sx.drawImage(st.c, tx * 64, ty * 64, tw, th, 0, 0, tw, th)
                            t = sx.getImageData(0, 0, tw, th)
                        }
                        catch (err) {
                            t = null //跨域污染：此後各塊皆同
                        }
                        st.tiles.set(k, t)
                    }
                    if (!t) {
                        return null
                    }
                    let lx = cx - tx * 64
                    let ly = cy - ty * 64
                    return { d: t.data, o: (ly * t.width + lx) * 4, t, lx, ly }
                }
                //像素有墨跡：alpha 不足(透明處，見到的是其下頁面)不算；背景眾數透明者(未繪製處透明之圖表)以 alpha 判定，
                //背景眾數有色者(含半透明或不透明底)以與背景之色差/alpha 差判定(同 canvasInkRects 之 alpha/bg 判準)
                let gCanvasInk = (st, cx, cy) => {
                    let px = gCanvasRead(st, cx, cy)
                    if (!px) {
                        return false
                    }
                    let { d, o, t, lx, ly } = px
                    if (d[o + 3] < 40) {
                        return false
                    }
                    //背景色取該像素所在 16x16 小格之眾數(各色量化至 8 階後計數；漸層底、大片均勻底之任一處皆以其周遭為背景，不受整張畫布之色分布影響)
                    let cell = (ly >> 4) * 4 + (lx >> 4)
                    t.modes = t.modes || []
                    let mode = t.modes[cell]
                    if (!mode) {
                        let cnt = new Map()
                        let bestN = 0
                        for (let yy = (ly >> 4) << 4; yy < Math.min(t.height, ((ly >> 4) + 1) << 4); yy++) {
                            for (let xx = (lx >> 4) << 4; xx < Math.min(t.width, ((lx >> 4) + 1) << 4); xx++) {
                                let p = (yy * t.width + xx) * 4
                                let key = ((d[p] >> 3) << 15) | ((d[p + 1] >> 3) << 10) | ((d[p + 2] >> 3) << 5) | (d[p + 3] >> 3)
                                let n = (cnt.get(key) || 0) + 1
                                cnt.set(key, n)
                                if (n > bestN) {
                                    bestN = n
                                    mode = [d[p], d[p + 1], d[p + 2], d[p + 3]]
                                }
                            }
                        }
                        t.modes[cell] = mode
                    }
                    if (mode[3] < 40) {
                        return true //背景為透明(未繪製處透明之圖表)：有 alpha 即墨跡
                    }
                    return Math.max(Math.abs(d[o] - mode[0]), Math.abs(d[o + 1] - mode[1]), Math.abs(d[o + 2] - mode[2]), Math.abs(d[o + 3] - mode[3])) > 24
                }
                //視窗座標(x, y)對應之 canvas 位圖座標；落在內容盒之外(或 object-fit 之留白處)回 null
                let gCanvasPoint = (st, x, y) => {
                    let m = st.map
                    if (x < m.ox || x >= m.ox + m.dw || y < m.oy || y >= m.oy + m.dh) {
                        return null
                    }
                    return [Math.min(st.w - 1, Math.max(0, Math.floor((x - m.ox) * m.sx))), Math.min(st.h - 1, Math.max(0, Math.floor((y - m.oy) * m.sy)))]
                }
                //graphicAt(x, y, fx, fr, region)：取樣點(左上角整數座標 x, y)之圖形內容；fx、fr 為浮層(fixed 元素與其矩形)；region 為候選範圍
                let graphicAt = (x, y, fx, fr, region) => {
                    let ctx = gCtxOf(region)
                    let inRect = (r) => x + 0.5 >= r.left && x + 0.5 < r.right && y + 0.5 >= r.top && y + 0.5 < r.bottom
                    //遮蓋判定(stack 為該點之命中堆疊，自上而下)：排在「圖形自身或其祖先」之前者即疊在其上；視為蓋住者——有自身底色(含遮罩之淡底，alpha 乘累積 opacity > 0.05)、
                    //背景圖、img/video/iframe、位圖在該點不透明之可讀 canvas(不可讀者保守視為蓋住)、不透明填色(alpha ≥ 0.9)之 svg 圖形(白色遮片；彩色者本身即內容，輪到它時另判)；
                    //完全透明或 opacity 近 0 之層(文字標籤、透明點擊攔截層)不算(穿透，同舊規則)
                    //命中堆疊取於整數座標 (x, y)：document.elementsFromPoint 對 HTML 方塊之命中區為 [x, x+1) x [y, y+1)(實測：小數座標之區塊向右下延伸，
                    //x+0.5 會把右/下一像素之元素也算進來，如緊貼浮層左緣之圖形被判成浮層蓋住)，即本像素；svg 圖形之命中則是精確幾何，另以實際命中之取樣點 at 取 fine 堆疊，
                    //供裁切確認(細於 1px 之線在整數點上可能不命中)與 svg 遮片之疊序判定
                    let occluded = (el, stack, fine) => {
                        let inFine = fine.includes(el)
                        for (let h of stack) {
                            if (h === el || h.contains(el)) {
                                break
                            }
                            if (h.ownerSVGElement || h.tagName.toLowerCase() === 'svg') {
                                continue //svg 內之圖形見下
                            }
                            let hcs = window.getComputedStyle(h)
                            let hop = gUpOf(h).op
                            if (hop < G_MIN_ALPHA) {
                                continue
                            }
                            if (h.tagName === 'CANVAS') {
                                let hst = gCanvasOf(ctx, h)
                                let hp = hst ? gCanvasPoint(hst, x + 0.5, y + 0.5) : null
                                let hpx = hp ? gCanvasRead(hst, hp[0], hp[1]) : null
                                if (!hpx || hpx.d[hpx.o + 3] * hop >= 230) {
                                    return true
                                }
                                continue
                            }
                            if (['IMG', 'VIDEO', 'IFRAME'].includes(h.tagName) || gColor(hcs.backgroundColor).a * hop > 0.05 || (hcs.backgroundImage && hcs.backgroundImage !== 'none')) {
                                return true
                            }
                        }
                        //不透明填色之 svg 圖形(白色遮片)：圖形自身在命中堆疊內者，堆疊順序即疊序(排在前者在上，含他根之遮片)；圖形不在堆疊內(pointer-events:none)
                        //則疊序不明，僅認同一 svg 內文件順序居後者(svg 圖形之疊序即文件順序)
                        for (let h of fine) {
                            if (h === el || h.contains(el)) {
                                break
                            }
                            if (!h.ownerSVGElement) {
                                continue
                            }
                            let hcs = window.getComputedStyle(h)
                            let hop = gUpOf(h).op
                            if (hop >= G_MIN_ALPHA && gPaint(hcs.fill, Number(hcs.fillOpacity), hop) >= 0.9 && !hcs.fill.startsWith('url(') && (inFine || (h.ownerSVGElement === el.ownerSVGElement && !!(el.compareDocumentPosition(h) & Node.DOCUMENT_POSITION_FOLLOWING)))) {
                                return true
                            }
                        }
                        return false
                    }
                    //q：命中內容之矩形(預設點大小：nearestOutside 取 max(top, y-1) 等，與舊規則命中元素之 1px 守距一致)；at：實際命中之取樣點(svg 圖形用之；其餘為 (x, y))
                    let hit = (d, el, clip, q, at) => {
                        let stack = document.elementsFromPoint(x, y)
                        if (stack.some((h) => fx.contains(h))) {
                            return 'covered' //浮層蓋著此點(面板可大於 fixed 外層)：其後之內容看不見
                        }
                        let fine = at[0] === x && at[1] === y ? stack : document.elementsFromPoint(at[0], at[1])
                        if ((clip && !fine.includes(el)) || occluded(el, stack, fine)) {
                            return null
                        }
                        gNote(d)
                        return q || { left: x - 1, top: y - 1, right: x + 1, bottom: y + 1 }
                    }
                    let center = [x + 0.5, y + 0.5]
                    let corner = [x, y]
                    //使用端宣告之繪圖區：整區(與其裁切範圍之交集)視為內容
                    for (let a of ctx.areas) {
                        if (fx.contains(a) || !gVisible(a)) {
                            continue
                        }
                        let ar = a.getBoundingClientRect()
                        if (inRect(rectInter({ left: ar.left, top: ar.top, right: ar.right, bottom: ar.bottom }, gUpOf(a).box))) {
                            let r = hit(`area:${gDesc(a)}`, a, gUpOf(a).clip && window.getComputedStyle(a).pointerEvents !== 'none', null, corner)
                            if (r) {
                                return r === 'covered' ? null : r
                            }
                        }
                    }
                    //svg 圖形
                    for (let root of ctx.roots) {
                        if (fx.contains(root)) {
                            continue
                        }
                        for (let s of gShapesOf(ctx, root, region)) {
                            if (x + 1 < s.q.left || x > s.q.right || y + 1 < s.q.top || y > s.q.bottom || fx.contains(s.el) || (s.box && !inRect(s.box))) {
                                continue
                            }
                            if (s.isText) {
                                //字行框：取樣點(左上角)落在框內(同舊規則 textRectAt 之含邊界判定)，回傳該框
                                let tr = s.rects.find((q) => x >= q.left && x <= q.right && y >= q.top && y <= q.bottom)
                                if (tr) {
                                    let r = hit(`svg:${gDesc(s.el)}`, s.el, s.clip, tr, center)
                                    if (r) {
                                        return r === 'covered' ? null : r
                                    }
                                }
                                continue
                            }
                            //取樣足跡：像素中心加其上下左右各距中心 0.45px 之四點(皆在本像素內)——細於 1px 之線落在像素內不致漏掉，又不越過像素邊界把相鄰一列(行)像素之內容算進本列(緊貼浮層邊之平行線不致誤退左右(上下)邊)
                            for (let [px, py] of [[x + 0.5, y + 0.5], [x + 0.05, y + 0.5], [x + 0.95, y + 0.5], [x + 0.5, y + 0.05], [x + 0.5, y + 0.95]]) {
                                let p = new DOMPoint(px, py).matrixTransform(s.inv)
                                if ((s.strokeOK && s.el.isPointInStroke(p)) || (s.fillOK && s.el.isPointInFill(p))) {
                                    let r = hit(`svg:${gDesc(s.el)}`, s.el, s.clip, null, [px, py])
                                    if (r) {
                                        return r === 'covered' ? null : r
                                    }
                                    break
                                }
                            }
                        }
                    }
                    //canvas 像素
                    for (let c of ctx.canvases) {
                        if (fx.contains(c)) {
                            continue
                        }
                        let st = gCanvasOf(ctx, c)
                        if (!st || (st.box && !inRect(st.box))) {
                            continue
                        }
                        let cp = gCanvasPoint(st, x + 0.5, y + 0.5)
                        if (cp && gCanvasInk(st, cp[0], cp[1])) {
                            let rr = hit(`canvas:${gDesc(c)}`, c, st.clip, null, corner)
                            if (rr) {
                                return rr === 'covered' ? null : rr
                            }
                        }
                    }
                    return null
                }
                //舊規則(HTML 內容)：點在視窗外、浮層外層矩形內者無內容；其餘取命中堆疊交 legacyContentAt
                let contentAt = (x, y, fx, fr) => {
                    let vw = window.innerWidth
                    let vh = window.innerHeight
                    if (x < 0 || y < 0 || x >= vw || y >= vh) {
                        return null
                    }
                    if (x >= fr.left && x < fr.right && y >= fr.top && y < fr.bottom) {
                        return null
                    }
                    return legacyContentAt(x, y, fx, document.elementsFromPoint(x, y))
                }
                //舊規則本體(1.0.9 之 contentAt，逐字不改；圖形規則不得改變其回傳)：自命中堆疊逐元素沿祖先找文字行框或有可見邊界之小元件，
                //遇祖先寬 > 480 或高 > 120 之大容器即止(此即大型 svg/canvas 內圖形之盲區，由 graphicAt 補)
                let legacyContentAt = (x, y, fx, stack) => {
                    let vw = window.innerWidth
                    let vh = window.innerHeight
                    for (let h of stack) {
                        if (fx.contains(h)) {
                            return null //浮層自身蓋著此點(面板可大於 fixed 外層)：其後之頁面內容看不見，不算
                        }
                        for (let a = h; a && a !== document.body && a !== document.documentElement; a = a.parentElement) {
                            let t = textRectAt(a, x, y)
                            if (t) {
                                return t
                            }
                            let ar = a.getBoundingClientRect()
                            if (ar.width > 480 || ar.height > 120) {
                                break
                            }
                            if (hasBoundary(a)) {
                                return ar
                            }
                        }
                        let hr = h.getBoundingClientRect()
                        let hcs = window.getComputedStyle(h)
                        if (hcs.position === 'fixed' && hr.width >= vw * 0.9 && hr.height >= vh * 0.9 && rgbaOf(hcs.backgroundColor)[3] === 0) {
                            continue //全畫面透明層：往下找其後之頁面內容
                        }
                        return null
                    }
                    return null
                }
                //目標四周(自其可見範圍向外，掃至預設框線外緣外 2px)、浮層範圍外最近之可見內容；每邊沿線每 4px 取一點，回傳各邊內容之對邊位置(無者 null)。
                //內容一部分被浮層蓋住時，以浮層外第一個可見點為界。另回傳 g：圖形(svg 圖形、canvas 像素、宣告區)之各邊最近內容(見下方圖形掃描)，
                //與 HTML 內容分開記——兩者之退入規則不同(HTML 取間隙正中、圖形取剛好不蓋)
                let nearestOutside = (inner, box, fx, fr) => {
                    let res = { top: null, bottom: null, left: null, right: null }
                    let lim = edge + strokeHalf + 2
                    let along = (a0, a1) => {
                        let arr = []
                        for (let a = a0; a < a1; a += 4) {
                            arr.push(a)
                        }
                        arr.push(a1)
                        return arr
                    }
                    //沿線取樣內縮 1px：端點若恰落在浮層邊線上(內容撐滿面板寬時 inner 邊 = 浮層邊)，會判成浮層外、誤把緊貼浮層側邊之
                    //背景內容當成上下方內容(tokens cht 權限浮層下緣框線切過末項文字殷鑑)；elementsFromPoint 會把座標取整，內縮 0.5px 不夠
                    for (let y of along(inner.top + 1, inner.bottom - 1)) {
                        for (let x = Math.floor(inner.right) + 1; x <= box.right + lim; x++) {
                            let q = contentAt(x, y, fx, fr)
                            if (q) {
                                let p = Math.max(q.left, x - 1)
                                res.right = res.right === null ? p : Math.min(res.right, p)
                                break
                            }
                        }
                        for (let x = Math.ceil(inner.left) - 1; x >= box.left - lim; x--) {
                            let q = contentAt(x, y, fx, fr)
                            if (q) {
                                let p = Math.min(q.right, x + 1)
                                res.left = res.left === null ? p : Math.max(res.left, p)
                                break
                            }
                        }
                    }
                    for (let x of along(inner.left + 1, inner.right - 1)) {
                        for (let y = Math.floor(inner.bottom) + 1; y <= box.bottom + lim; y++) {
                            let q = contentAt(x, y, fx, fr)
                            if (q) {
                                let p = Math.max(q.top, y - 1)
                                res.bottom = res.bottom === null ? p : Math.min(res.bottom, p)
                                break
                            }
                        }
                        for (let y = Math.ceil(inner.top) - 1; y >= box.top - lim; y--) {
                            let q = contentAt(x, y, fx, fr)
                            if (q) {
                                let p = Math.min(q.bottom, y + 1)
                                res.top = res.top === null ? p : Math.max(res.top, p)
                                break
                            }
                        }
                    }
                    //圖形(svg 圖形、canvas 像素、宣告區)：資料線可橫貫浮層下方任一處，不只在文字內容之正下方，故沿**框線全長**取樣
                    //(框矩形邊的全長，逐 px)，而非只取內容範圍 inner 之寬高(舊規則：晶片、按鈕等鄰項緊貼文字列才有意義)。
                    //只掃框線帶本身(框矩形邊外 1 至 BOX_PAD px，四向同一算法：像素列/行與帶相交者)，不自內容邊起——框矩形內(含 inkPad 區)之內容框線蓋不到，
                    //讓它介入只會使框線退入文字；框線四角之 BOX_PAD x BOX_PAD 小方格不掃——角落旁側之內容屬左右(上下)兩邊之事，
                    //掃角落會使平行於浮層邊之橫(縱)線誤退左右(上下)邊，而落在角落之孤立點至多被圓角描邊蓋住數 px²(已知限制)
                    if (graphics) {
                        let glim = edge + strokeHalf //圖形只在框線帶(外緣)內才介入：不像舊規則另留 2px，免得距框線外緣數 px 之淡格線等使既有標準圖無謂改動
                        //候選範圍：浮層外層矩形與框矩形之聯集外擴 64px(取整至 128px 格，同範圍之多個目標共用候選清單)；面板可大於 fixed 外層(0x0 之外層包 absolute 面板)，
                        //框矩形不在外層矩形內，故須計入
                        let gReg = {
                            left: Math.floor((Math.min(fr.left, box.left) - 64) / 128) * 128,
                            top: Math.floor((Math.min(fr.top, box.top) - 64) / 128) * 128,
                            right: Math.ceil((Math.max(fr.right, box.right) + 64) / 128) * 128,
                            bottom: Math.ceil((Math.max(fr.bottom, box.bottom) + 64) / 128) * 128
                        }
                        let gAt = (x, y) => {
                            if (x < 0 || y < 0 || x >= window.innerWidth || y >= window.innerHeight || (x >= fr.left && x < fr.right && y >= fr.top && y < fr.bottom)) {
                                return null
                            }
                            try {
                                return graphicAt(x, y, fx, fr, gReg)
                            }
                            catch (err) {
                                return null
                            }
                        }
                        let gres = res.g = { top: null, bottom: null, left: null, right: null }
                        let rec = (side, p, x, y, nearer) => {
                            if (gres[side] === null || nearer(p, gres[side])) {
                                gres[side] = p
                                if (debug) {
                                    gnear[side] = `${gLast}@${x},${y}`
                                }
                            }
                        }
                        //沿邊逐 px 取樣(舊規則 4px)：1px 細線(垂直線、柱、影線)落在像素格線上，取樣間距大於 1px 會整條漏掉
                        let lt = (a, b) => a < b
                        let gt = (a, b) => a > b
                        //框線帶為框矩形邊外 [1, glim]：下/右帶 [邊+1, 邊+glim]、上/左帶 [邊-glim, 邊-1]。掃描含緊鄰矩形邊之一列(行)作守距——與框線內緣相切之
                        //內容不被蓋住，但延伸至框線四角之縱(橫)向內容(貼著浮層側邊、伸到浮層下方之柱，其在角落處與框線相交)由此一列(行)攔下；
                        //下/右 y 自 floor(邊) 至 < 邊+glim；上/左 y 自 ceil(邊)-1 至 ≥ floor(邊-glim)(邊為小數時，帶端點所在列也算)。沿邊取框矩形全長之像素列(行)
                        for (let x = Math.floor(box.left); x < Math.ceil(box.right); x++) {
                            for (let y = Math.floor(box.bottom); y < box.bottom + glim; y++) {
                                let q = gAt(x, y)
                                if (q) {
                                    rec('bottom', Math.max(q.top, y - 1), x, y, lt)
                                    break
                                }
                            }
                            for (let y = Math.ceil(box.top) - 1; y >= Math.floor(box.top - glim); y--) {
                                let q = gAt(x, y)
                                if (q) {
                                    rec('top', Math.min(q.bottom, y + 1), x, y, gt)
                                    break
                                }
                            }
                        }
                        for (let y = Math.floor(box.top); y < Math.ceil(box.bottom); y++) {
                            for (let x = Math.floor(box.right); x < box.right + glim; x++) {
                                let q = gAt(x, y)
                                if (q) {
                                    rec('right', Math.max(q.left, x - 1), x, y, lt)
                                    break
                                }
                            }
                            for (let x = Math.ceil(box.left) - 1; x >= Math.floor(box.left - glim); x--) {
                                let q = gAt(x, y)
                                if (q) {
                                    rec('left', Math.min(q.right, x + 1), x, y, gt)
                                    break
                                }
                            }
                        }
                    }
                    return res
                }
                //框線不得壓到相鄰項目：自元素往上(至多 4 層)找「緊鄰兄弟」，框線(外擴 edge 後之描邊中心)置於兩者**可見範圍**間隙正中——
                //以可見範圍而非元素框計(元素框含透明內距：圖示鈕之內距使中點貼回目標文字，PERM 編輯模式開關殷鑑)；無可見內容之兄弟不算鄰項；
                //四向各自取最近一層有鄰項者(左側鄰項在本層、右側鄰項在上層時兩側皆夾)；兩者可見範圍重疊者無間隙可置，不夾；相距遠者之中點本就在框外，不影響
                let clampByNeighbors = (e, rc) => {
                    let { left, top, right, bottom } = rc
                    let own = visExtent(e) || e.getBoundingClientRect()
                    let done = { top: false, bottom: false, left: false, right: false }
                    let node = e
                    for (let lv = 0; lv < 4 && node && node.parentElement; lv++) {
                        let nr = node.getBoundingClientRect()
                        let hit = { top: false, bottom: false, left: false, right: false }
                        for (let s of node.parentElement.children) {
                            if (s === node) {
                                continue
                            }
                            let sc = window.getComputedStyle(s)
                            if (sc.display === 'none' || sc.visibility === 'hidden' || Number(sc.opacity) === 0) {
                                continue
                            }
                            let sr = s.getBoundingClientRect()
                            if (sr.width <= 0 || sr.height <= 0) {
                                continue
                            }
                            let hOverlap = sr.right > nr.left && sr.left < nr.right
                            let vOverlap = sr.bottom > nr.top && sr.top < nr.bottom
                            let dir = null
                            if (hOverlap && sr.bottom <= nr.top + 1) {
                                dir = 'top'
                            }
                            else if (hOverlap && sr.top >= nr.bottom - 1) {
                                dir = 'bottom'
                            }
                            else if (vOverlap && sr.right <= nr.left + 1) {
                                dir = 'left'
                            }
                            else if (vOverlap && sr.left >= nr.right - 1) {
                                dir = 'right'
                            }
                            if (!dir || done[dir]) {
                                continue
                            }
                            let v = visExtent(s)
                            if (!v) {
                                continue //無可見內容之間隔元素：框線可經過
                            }
                            hit[dir] = true
                            if (dir === 'top' && v.bottom <= own.top) {
                                top = Math.max(top, (v.bottom + own.top) / 2 + edge)
                            }
                            else if (dir === 'bottom' && v.top >= own.bottom) {
                                bottom = Math.min(bottom, (own.bottom + v.top) / 2 - edge)
                            }
                            else if (dir === 'left' && v.right <= own.left) {
                                left = Math.max(left, (v.right + own.left) / 2 + edge)
                            }
                            else if (dir === 'right' && v.left >= own.right) {
                                right = Math.min(right, (own.right + v.left) / 2 - edge)
                            }
                        }
                        for (let d of Object.keys(hit)) {
                            if (hit[d]) {
                                done[d] = true
                            }
                        }
                        if (done.top && done.bottom && done.left && done.right) {
                            break
                        }
                        node = node.parentElement
                    }
                    //浮層(自身或祖先為 position:fixed，例：teleport 之下拉清單、浮出面板)：浮層外之頁面內容不是其 DOM 兄弟，改以命中測試
                    //找目標四周、浮層範圍外最近之可見內容，框線置於與其間隙正中；浮層本身為目標時以其內容之可見範圍計(框線可退入浮層內距)。
                    //四周無內容者照常外擴(一律畫在浮層內側會讓內距小之清單框線貼字：PERM 時間清單下緣 2px、關聯模式清單左緣 1px 殷鑑)；
                    //全畫面之浮層(彈窗、抽屜之外層)不適用
                    let fx = null
                    let beforeGraphics = null //圖形規則介入前之夾制結果(圖形夾到沒有尺寸時退回此值，不連同 HTML 規則之夾制一併撤銷)
                    for (let a = e; a && a !== document.body && a !== document.documentElement; a = a.parentElement) {
                        if (window.getComputedStyle(a).position === 'fixed') {
                            fx = a
                            break
                        }
                    }
                    let vw = window.innerWidth
                    let vh = window.innerHeight
                    let fr = fx ? fx.getBoundingClientRect() : null
                    if (fx && !(fr.width >= vw * 0.9 && fr.height >= vh * 0.9)) {
                        //目標涵蓋整個浮層(浮層本身、其透明外層包裝或撐滿之面板)：以撐滿浮層之有邊界面板的內容計，框線可退入面板內距
                        //(元件庫之浮層多為「透明之 fixed 外層 + 有底色陰影之內層面板」，只認 fixed 元素本身會漏掉：PERM 語系清單框線壓觸發區殷鑑)
                        let inner = own
                        if (e === fx || sameRect(own, fr, 2)) {
                            let panel = boundedFilling(e, own, 2)
                            if (panel) {
                                inner = visExtent(panel, true, panel.getBoundingClientRect()) || own
                            }
                        }
                        let near = nearestOutside(inner, { left, top, right, bottom }, fx, fr)
                        if (debug) {
                            dbg.push({ inner: [inner.left, inner.top, inner.right, inner.bottom].map((n) => Math.round(n * 10) / 10), near: { top: near.top, bottom: near.bottom, left: near.left, right: near.right }, graphic: gnear })
                            gnear = {}
                        }
                        if (near.top !== null && near.top <= inner.top) {
                            top = Math.max(top, (near.top + inner.top) / 2 + edge)
                        }
                        if (near.bottom !== null && near.bottom >= inner.bottom) {
                            bottom = Math.min(bottom, (inner.bottom + near.bottom) / 2 - edge)
                        }
                        if (near.left !== null && near.left <= inner.left) {
                            left = Math.max(left, (near.left + inner.left) / 2 + edge)
                        }
                        if (near.right !== null && near.right >= inner.right) {
                            right = Math.min(right, (inner.right + near.right) / 2 - edge)
                        }
                        //圖形(資料線、圖例、畫布像素)：不似 HTML 鄰項取間隙正中——圖形遍布整個繪圖區，浮層又常比其文字寬得多，取正中會使框線縮到只剩文字範圍、
                        //不再框住浮層；改為框線外緣退到內容前 1px(剛好不蓋)，且框內緣不越過文字；內容貼近文字、放不下 5px 描邊時同舊規則取間隙正中
                        let gn = near.g
                        if (gn) {
                            beforeGraphics = { left, top, right, bottom }
                            let fo = edge + strokeHalf //框線外緣距框矩形邊(BOX_PAD)
                            if (gn.top !== null && gn.top <= inner.top) {
                                let t = gn.top + fo + 1
                                top = Math.max(top, t <= inner.top + 1 ? t : (gn.top + inner.top) / 2 + edge)
                            }
                            if (gn.bottom !== null && gn.bottom >= inner.bottom) {
                                let b = gn.bottom - fo
                                bottom = Math.min(bottom, b >= inner.bottom - 1 ? b : (inner.bottom + gn.bottom) / 2 - edge)
                            }
                            if (gn.left !== null && gn.left <= inner.left) {
                                let l = gn.left + fo + 1
                                left = Math.max(left, l <= inner.left + 1 ? l : (gn.left + inner.left) / 2 + edge)
                            }
                            if (gn.right !== null && gn.right >= inner.right) {
                                let r = gn.right - fo
                                right = Math.min(right, r >= inner.right - 1 ? r : (inner.right + gn.right) / 2 - edge)
                            }
                        }
                    }
                    if (debug) {
                        let rr = (q) => (q ? [q.left, q.top, q.right, q.bottom].map((n) => Math.round(n * 10) / 10) : null)
                        dbg.push({ el: `${e.tagName}.${String(e.className || '').slice(0, 30)}`, own: rr(own), fixed: fx ? `${fx.tagName}.${String(fx.className || '').slice(0, 30)}` : null, fr: rr(fr), rcIn: rr(rc), out: rr({ left, top, right, bottom }) })
                    }
                    if (beforeGraphics && (right - left < 1 || bottom - top < 1)) {
                        //圖形規則使框夾到沒有尺寸：只撤銷圖形之夾制，HTML 規則之結果保留
                        left = beforeGraphics.left
                        top = beforeGraphics.top
                        right = beforeGraphics.right
                        bottom = beforeGraphics.bottom
                    }
                    if (right - left < 1 || bottom - top < 1) {
                        return rc //防呆：夾到沒有尺寸則不夾
                    }
                    return { left, top, right, bottom }
                }
                let rectOf = (e) => {
                    let r = e.getBoundingClientRect()
                    if (!fit) {
                        return r
                    }
                    if (hasBoundary(e)) {
                        return clampByNeighbors(e, r) //元素自身有可見邊界(晶片、按鈕、卡片、底色列)：量元素本身
                    }
                    //無可見邊界者(透明容器、整行文字、開關/勾選列等)：量其可見內容(有邊界之子元素框與文字行框之聯集)並一律外擴 inkPad——
                    //勾選框、開關軌道、圖示、輸入框等子元素與文字同留白(原只外擴文字、子元素框不外擴，框線貼勾選框與開關軌道 0–1px：
                    //tokens 權限列、編輯模式開關殷鑑)；透明子元素之內距不計入(不可見)
                    let v = visExtent(e, true)
                    if (!v) {
                        let rg = document.createRange()
                        rg.selectNodeContents(e)
                        let c = rg.getBoundingClientRect()
                        return clampByNeighbors(e, (c.width > 0 && c.height > 0) ? c : r)
                    }
                    //可見內容恰為單一有邊界之後代(透明外層包著之面板、按鈕)：視同框該元素本身，不外擴(同直接框有邊界之元素)
                    if (boundedFilling(e, v, 1)) {
                        return clampByNeighbors(e, v)
                    }
                    return clampByNeighbors(e, { left: v.left - inkPad, top: v.top - inkPad, right: v.right + inkPad, bottom: v.bottom + inkPad })
                }
                let box = null
                if (within) {
                    let scope = document.querySelector(within)
                    if (!scope) {
                        return null
                    }
                    box = scope.getBoundingClientRect()
                }
                else {
                    box = { left: 0, top: 0, right: window.innerWidth, bottom: window.innerHeight }
                }
                let parts = []
                els.forEach((e) => {
                    let r = rectOf(e)
                    if (!(r.right - r.left > 0) || !(r.bottom - r.top > 0)) {
                        return
                    }
                    let cs = window.getComputedStyle(e)
                    if (cs.display === 'none' || cs.visibility === 'hidden' || Number(cs.opacity) === 0) {
                        return
                    }
                    if (r.bottom <= box.top || r.top >= box.bottom || r.right <= box.left || r.left >= box.right) {
                        return
                    }
                    parts.push(r)
                })
                if (parts.length === 0) {
                    return null
                }
                let left = Math.max(Math.min(...parts.map((r) => r.left)), box.left)
                let top = Math.max(Math.min(...parts.map((r) => r.top)), box.top)
                let right = Math.min(Math.max(...parts.map((r) => r.right)), box.right)
                let bottom = Math.min(Math.max(...parts.map((r) => r.bottom)), box.bottom)
                if (right <= left || bottom <= top) {
                    return null
                }
                let res = { x: left, y: top, width: right - left, height: bottom - top }
                if (debug) {
                    res._dbg = dbg
                }
                return res
            }
            //edge：目標矩形邊至框線中心之距離(外擴 BOX_PAD、描邊 BOX_STROKE 置於外擴區內側)；
            //E2E_BOX_DEBUG=1 時印出各元素之可見範圍、浮層矩形、內容範圍、浮層外最近內容與夾邊前後(審圖查因用)
            let debug = !!process.env.E2E_BOX_DEBUG
            let res = await loc.evaluateAll(measureEls, { within, fit, inkPad, edge: BOX_PAD - BOX_STROKE / 2, strokeHalf: BOX_STROKE / 2, debug, graphics, areaSels })
            if (res && res._dbg) {
                console.log(`[itemsUnionBox debug] ${isLoc ? 'Locator' : sel} ${JSON.stringify(res._dbg)} → ${JSON.stringify([res.x, res.y, res.x + res.width, res.y + res.height])}`)
                delete res._dbg
            }
            return res
        },
    }
